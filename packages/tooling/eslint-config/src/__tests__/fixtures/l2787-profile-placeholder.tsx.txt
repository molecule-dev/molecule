import { useEffect, useState } from 'react'
import { Link, useParams, useSearchParams } from 'react-router'

import { useQuery } from '@molecule/app-query-react'
import { useAuth } from '@molecule/app-react'

import { ArticlePreview } from '../components/ArticlePreview.js'
import { Pagination } from '../components/Pagination.js'
import type { ConduitUser } from '../config.js'
import { avatarUrl } from '../lib/format.js'
import { useLabel } from '../lib/i18n.js'
import { articlesPage } from '../resources/articles.js'
import { profiles } from '../resources/profiles.js'

const PAGE_SIZE = 10

/** A reader's profile: who they are, and the articles they wrote or favorited. */
export function Profile({ favorites = false }: { favorites?: boolean }) {
  const label = useLabel()
  const { username = '' } = useParams<{ username: string }>()
  const [search] = useSearchParams()
  const { user } = useAuth<ConduitUser>()
  const viewer = user?.username ?? 'anonymous'
  const page = Math.max(1, Number.parseInt(search.get('page') ?? '1', 10) || 1)
  const offset = (page - 1) * PAGE_SIZE

  const profile = useQuery(profiles.detail(viewer, username))
  const list = useQuery({
    ...articlesPage(viewer, favorites ? { favorited: username, limit: PAGE_SIZE, offset } : { author: username, limit: PAGE_SIZE, offset }),
    principal: viewer,
  })

  const [following, setFollowing] = useState(false)

  useEffect(() => {
    if (profile.data) setFollowing(profile.data.following)
  }, [profile.data])

  const toggleFollow = async () => {
    const next = !following
    setFollowing(next)
    try {
      const saved = await profiles.toggle(viewer, username, 'follow', next)
      setFollowing(saved.following)
    } catch (error) {
      console.error('Follow toggle failed:', error)
      setFollowing(!next)
    }
  }

  const isSelf = user?.username === username
  const rows = list.data?.articles ?? []
  const total = list.data?.articlesCount ?? 0
  const pages = Math.max(1, Math.ceil(total / PAGE_SIZE))
  const base = favorites ? `/profile/${username}/favorites` : `/profile/${username}`
  const href = (target: number): string => (target > 1 ? `${base}?page=${target}` : base)

  return (
    <div className="profile-page">
      <div className="user-info">
        <div className="container">
          <div className="row">
            <div className="col-xs-12 col-md-10 offset-md-1">
              {profile.data ? (
                <>
                  <img className="user-img" src={avatarUrl(profile.data.image)} alt={username} />
                  <h4>{profile.data.username}</h4>
                  <p>{profile.data.bio ?? ''}</p>
                  {isSelf ? (
                    <Link to="/settings" className="btn btn-sm btn-outline-secondary action-btn">
                      {label('profile.edit', 'Edit Profile Settings')}
                    </Link>
                  ) : (
                    <button
                      type="button"
                      className="btn btn-sm btn-outline-secondary action-btn"
                      onClick={toggleFollow}
                      data-mol-id="profile-follow"
                    >
                      {following
                        ? `${label('profile.unfollow', 'Unfollow')} ${username}`
                        : `${label('profile.follow', 'Follow')} ${username}`}
                    </button>
                  )}
                </>
              ) : (
                <p className="loading-note">{label('common.loading', 'Loading...')}</p>
              )}
            </div>
          </div>
        </div>
      </div>

      <div className="container">
        <div className="row">
          <div className="col-xs-12 col-md-10 offset-md-1">
            <div className="articles-toggle">
              <ul className="nav nav-pills outline-active">
                <li className="nav-item">
                  <Link
                    className={favorites ? 'nav-link' : 'nav-link active'}
                    to={`/profile/${username}`}
                    data-mol-id="profile-articles"
                  >
                    {label('profile.myArticles', 'My Articles')}
                  </Link>
                </li>
                <li className="nav-item">
                  <Link
                    className={favorites ? 'nav-link active' : 'nav-link'}
                    to={`/profile/${username}/favorites`}
                    data-mol-id="profile-favorites"
                  >
                    {label('profile.favorited', 'Favorited')}
                  </Link>
                </li>
              </ul>
            </div>

            {list.status === 'error' ? (
              <div className="error-messages">
                <p>{label('feed.loadFailed', 'Unable to load articles.')}</p>
              </div>
            ) : null}

            {list.data && rows.length === 0 ? (
              <div className="empty-feed-message">
                {label('profile.empty', 'No articles are here... yet.')}
              </div>
            ) : null}

            {rows.map((article) => (
              <ArticlePreview
                key={article.slug}
                article={article}
                onChange={() => {
                  void list.refetch()
                }}
              />
            ))}

            <Pagination page={Math.min(page, pages)} pages={pages} href={href} />
          </div>
        </div>
      </div>
    </div>
  )
}
