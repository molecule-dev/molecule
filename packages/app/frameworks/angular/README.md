```typescript
interface HttpClient {
  /**
   * Base URL for all requests.
   */
  baseURL: string

  /**
   * Default headers for all requests.
   */
  defaultHeaders: Record<string, string>

  /**
   * Makes a generic HTTP request. `data` is a field of THIS config object — the same body
   * `post(url, data, config)` takes as its second argument.
   */
  request<T = unknown>(config: FullRequestConfig): Promise<HttpResponse<T>>

  /**
   * Makes a GET request.
   */
  get<T = unknown>(url: string, config?: RequestConfig): Promise<HttpResponse<T>>

  /**
   * Makes a POST request. `data` is the request BODY, not a config object: the wire body is
   * `data` itself, so `post(url, { data: body })` sends `{"data":body}` — pass `body` directly.
   * A config (headers, params) is the THIRD argument.
   */
  post<T = unknown>(url: string, data?: unknown, config?: RequestConfig): Promise<HttpResponse<T>>

  /**
   * Makes a PUT request. `data` is the request BODY (second argument) — never `{ data: body }`,
   * which sends `{"data":body}`; a config is the THIRD argument.
   */
  put<T = unknown>(url: string, data?: unknown, config?: RequestConfig): Promise<HttpResponse<T>>

  /**
   * Makes a PATCH request. `data` is the request BODY (second argument) — never `{ data: body }`,
   * which sends `{"data":body}`; a config is the THIRD argument.
   */
  patch<T = unknown>(url: string, data?: unknown, config?: RequestConfig): Promise<HttpResponse<T>>

  /**
   * Makes a DELETE request.
   */
  delete<T = unknown>(url: string, config?: RequestConfig): Promise<HttpResponse<T>>

  /**
   * Adds a request interceptor.
   * Returns a function to remove the interceptor.
   */
  addRequestInterceptor(interceptor: RequestInterceptor): () => void

  /**
   * Adds a response interceptor.
   * Returns a function to remove the interceptor.
   */
  addResponseInterceptor(interceptor: ResponseInterceptor): () => void

  /**
   * Adds an error interceptor.
   * Returns a function to remove the interceptor.
   */
  addErrorInterceptor(interceptor: ErrorInterceptor): () => void

  /**
   * Sets the authorization token.
   *
   * A static token: every request carries it until the next `setAuthToken` call. A client
   * that has a {@link HttpClient.setAuthTokenResolver | resolver} (registered through
   * `@molecule/app-http`'s `setAuthTokenSource`) ignores this value when it assembles a
   * request — the resolver IS the session, and reading it per request is what keeps a
   * session adopted without an auth event from sending the previous account's bearer. Once
   * the resolver is removed, the static token is sent again.
   */
  setAuthToken(token: string | null): void

  /**
   * Returns the token the client applies to its next request: the resolver's answer when one
   * is registered, else the static token from `setAuthToken`.
   */
  getAuthToken(): string | null

  /**
   * Registers the LIVE token resolver every request of this client reads, or clears it with
   * `null`.
   *
   * `@molecule/app-http`'s `setAuthTokenSource()` calls this on every client it bonds, and the
   * client MUST consult the resolver again each time it assembles a request (and report it from
   * `getAuthToken()`). A session can change with no auth event and no observable state change —
   * `authClient.setSession()` with the same profile data, or the OAuth callback's
   * `setAccessToken()` — so a token pushed only at bond time or on `login`/`logout` events
   * leaves every later request carrying the PREVIOUS session's bearer while the UI shows the
   * new one.
   *
   * The resolved token takes the static token's place, with the same precedence: an
   * `Authorization` header passed per request or in the client's default headers wins over it
   * exactly where it won over a static token in this transport.
   *
   * A resolver that returns `null` (or `undefined`) means "this session has no body token": the
   * request carries no `Authorization` header of the client's own. A cookie-session API issues
   * no bearer, and the server prefers the header over the fresh cookie, so a kept bearer
   * authenticates as the prior — possibly deleted — session.
   *
   * Optional, so an implementation written against the previous interface still conforms.
   * `setAuthTokenSource()` follows such a client through a request interceptor instead, pushing
   * the source with `setAuthToken` before each request: that request carries the change only
   * when the transport applies its token after running its interceptors, so a transport that
   * applies it earlier sends a session adopted mid-page from the following request on.
   *
   * @param resolver - Returns the current access token (or null/undefined), or `null` to clear.
   */
  setAuthTokenResolver?(resolver: AuthTokenResolver | null): void

  /**
   * Registers a handler for authentication errors (401).
   *
   * @returns An unsubscribe function.
   */
  onAuthError(handler: () => void): () => void
}
```

```typescript
   *

  }
```

```typescript
   * Make a PATCH request.
   *
   * @param url - Request URL
   * @param data - Request body
   * @param config - Request configuration
   * @returns Observable of response data
   */
  patch<T>(url: string, data?: unknown, config?: RequestConfig): Observable<T> {
    return from(this.client.patch<T>(url, data, config)).pipe(map((res) => res.data))
  }

  /**
   * Make a DELETE request.
   *
   * @param url - Request URL
   * @param config - Request configuration
   * @returns Observable of response data
   */
  delete<T>(url: string, config?: RequestConfig): Observable<T> {
    return from(this.client.delete<T>(url, config)).pipe(map((res) => res.data))
  }

  /**
   * Get the full response (including headers, status, etc.).
   *
   * @param method - HTTP method
   * @param url - Request URL
   * @param config - Request configuration
   * @returns Observable of full response
   */
  request<T>(
    method: 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE',
    url: string,
    config?: RequestConfig,
  ): Observable<HttpResponse<T>> {
    const promise = (async () => {
      switch (method) {
        case 'GET':
          return this.client.get<T>(url, config)
        case 'POST':
          return this.client.post<T>(
            url,
            (config as RequestConfig & { data?: unknown })?.data,
            config,
          )
        case 'PUT':
          return this.client.put<T>(
            url,
            (config as RequestConfig & { data?: unknown })?.data,
            config,
          )
        case 'PATCH':
          return this.client.patch<T>(
            url,
            (config as RequestConfig & { data?: unknown })?.data,
            config,
          )
        case 'DELETE':
          return this.client.delete<T>(url, config)
      }
    })()

    return from(promise)
  }

  /**
   * Create a state subject for managing request state.
   *
   * @returns The created instance.
   */
  createState<T>(): BehaviorSubject<HttpState<T>> {
    return new BehaviorSubject<HttpState<T>>({
      data: null,
      loading: false,
      error: null,
    })
  }

  /**
   * Make a GET request with state management.
   *
   * @param url - Request URL
   * @param state - State subject to update
   * @param config - Request configuration
   */
  getWithState<T>(url: string, state: BehaviorSubject<HttpState<T>>, config?: RequestConfig): void {
    state.next({ ...state.value, loading: true, error: null })

    this.get<T>(url, config)
      .pipe(
        tap((data) => state.next({ data, loading: false, error: null })),
        catchError((error) => {
          state.next({ data: null, loading: false, error })
          throw error
        }),
      )
      .subscribe()
  }
}
```

#### `MoleculeStateService`

Wraps molecule state stores and exposes them as RxJS observables.

```typescript
@Injectable({
  providedIn: 'root',
})
/**
 * Molecule State Service class.
 */
export class MoleculeStateService implements OnDestroy {
  private subscriptions = new Map<Store<unknown>, () => void>()
  private subjects = new Map<Store<unknown>, BehaviorSubject<unknown>>()

  constructor(@Inject(STATE_PROVIDER) private provider: StateProvider) {}

  /**
   * Get an observable of the full store state.
   *
   * @param store - The store to observe
   * @returns Observable of the store state
   */
  observe<T>(store: Store<T>): Observable<T> {
    return this.getOrCreateSubject(store).asObservable() as Observable<T>
  }

  /**
   * Select a piece of state from a store.
   *
   * @param store - The store to select from
   * @param selector - Selector function
   * @returns Observable of the selected state
   */
  select<T, S>(store: Store<T>, selector: (state: T) => S): Observable<S> {
    return this.observe(store).pipe(map(selector), distinctUntilChanged())
  }

  /**
   * Get the current snapshot of a store's state.
   *
   * @param store - The store
   * @returns The result.
   */
  snapshot<T>(store: Store<T>): T {
    return store.getState()
  }

  /**
   * Update a store's state.
   *
   * @param store - The store to update
   * @param partial - Partial state or updater function
   */
  update<T>(store: Store<T>, partial: Partial<T> | ((state: T) => Partial<T>)): void {
    store.setState(partial)
  }

  /**
   * Gets or creates a BehaviorSubject for the given store.
   * @param store - The data store to observe.
   * @returns The BehaviorSubject tracking the store state.
   */
  private getOrCreateSubject<T>(store: Store<T>): BehaviorSubject<T> {
    let subject = this.subjects.get(store as Store<unknown>) as BehaviorSubject<T> | undefined

    if (!subject) {
      subject = new BehaviorSubject<T>(store.getState())
      this.subjects.set(store as Store<unknown>, subject as BehaviorSubject<unknown>)

      const unsubscribe = store.subscribe(() => {
        subject!.next(store.getState())
      })
      this.subscriptions.set(store as Store<unknown>, unsubscribe)
    }

    return subject
  }

  /**
   * Cleans up all store subscriptions and completes subjects on service destruction.
   */
  ngOnDestroy(): void {
    // Clean up all subscriptions
    this.subscriptions.forEach((unsubscribe) => unsubscribe())
    this.subscriptions.clear()

    // Complete all subjects
    this.subjects.forEach((subject) => subject.complete())
    this.subjects.clear()
  }
}
```

```typescript
   */

    const state = new BehaviorSubject<StorageValueState<T>>({
      value: defaultValue,
      loading: true,
      error: null,
    })

    // Load initial value
    this.get<T>(key).subscribe({
      next: (value) => {
        state.next({
          value: value ?? defaultValue,
          loading: false,
          error: null,
        })
      },
      error: (error) => {
        state.next({
          value: defaultValue,
          loading: false,
          error,
        })
      },
    })

    return state
  }

  /**
   * Update a reactive storage value.
   *
   * @param key - Storage key
   * @param value - New value
   * @param state - State subject to update
   */
  updateValue<T>(key: string, value: T, state: BehaviorSubject<StorageValueState<T>>): void {
    state.next({ ...state.value, loading: true })

    this.set(key, value).subscribe({
      next: () => {
        state.next({ value, loading: false, error: null })
      },
      error: (error) => {
```

#### `SessionReady`

Renders its content once `useAuth().status` has left `'loading'` (signed in,
signed out, or `'unavailable'`), and the `[slot=fallback]` content until
then. It never redirects. An unavailable session is unconfirmed and may
still have `user === null`; handle that status separately from sign-out.

After a full page load a returning visitor's session is restored by one
request. Until it answers, `useAuth().user` is `null` everywhere, so a
public page that renders a signed-in action at once lets that action run
against "signed out": a Follow button whose handler starts with
`if (!user) return` ignores the click, one that redirects to `/login` signs
the visitor out of the flow. Wrapping the routes (or passing `awaitSession`
to `provideMolecule`) makes every page's first render read the restored user
when restoration succeeds.

A visitor with no session to restore settles in microtasks without a
request. Must render inside the auth registration (`provideMolecule` with
`auth`). A client that nobody initialized starts its restore on this first
read. The bundled JWT client bounds each auth request (default 10 seconds)
and settles a timeout as `'unavailable'`. Custom clients must also settle
their initial session status.

Route-level redirects for signed-in-only pages stay with the app's own
guards.

```typescript
@Component({
  selector: 'mol-session-ready',
  standalone: true,
  template: `
    @if (status() === 'loading') {
      <ng-content select="[slot=fallback]" />
    } @else {
      <ng-content />
    }
  `,
})
export class SessionReady {
  /** The auth status this component holds on (`'loading'` shows the fallback). */
  readonly status = useAuth().status
}
```

#### `useVerifyPaymentReturn(options)`

purchase server-side with `POST /users/:id/verify-payment/:provider`. Call

plan is never granted. The request this hook makes is a same-origin call

is restored from the httpOnly cookie via `GET /users/me`.

transaction), and this hook additionally runs at most once per transaction
id per page. Safe on pages that are also reached normally: with no id in the
URL it stays `idle` and issues no request.

purchase still refreshes the cached profile (the plan is paid for), but the
hook calls neither `onVerified` nor `onError` and leaves the URL alone: by
then the address may belong to another page. A failure after unmount is
logged instead.

- Peer requirements: Angular 22 (`@angular/core` is pinned to 22.0.0) and
  rxjs 7.8+.
- Translations: import `t` from THIS package, not from
  `@molecule/app-i18n`. The re-exported `t` reads an internal Angular
  signal, so template bindings that CALL it (expose it on the component as
  above) re-evaluate automatically when the locale changes; the plain
  app-i18n `t` — or a one-time field assignment like `title = t(...)` —
  renders once and goes stale. The signal is bumped by `provideMolecule` —
  pass your `i18n` provider there (or call `bumpLocaleVersion()` from your
  own locale-change hook) for the reactivity to fire.
- `provideMolecule` only registers the providers you pass; injecting a
  Molecule service whose token was never provided fails at DI time.
  Per-concern helpers (`provideAuth`, `provideTheme`, ...) exist for
  piecemeal setup.
