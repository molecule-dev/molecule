# @molecule/app-drag-drop-dndkit-angular

Angular sortable list for `@molecule/app-drag-drop-dndkit`.

{@link SortableList} (`[molSortableList]`) renders a list the user reorders
with the pointer, touch or the keyboard, on `@dnd-kit/dom`. Each row is a
{@link SortableItem} (`[molSortableItem]`) — or a component calling
{@link useSortableItem} in its field initializer. The options are
`@molecule/app-drag-drop-dndkit`'s `SortableListOptions`, the same in the
React, Vue, Solid and Svelte bindings, so a list swaps between frameworks
unchanged. The row directives {@link SortableItem} and
{@link SortableItemHandle} are Angular's own: a row's id comes from a
template binding, which `useSortableItem` (read once, when the component is
constructed) cannot follow. The callbacks are inputs, not outputs, so a consumer passes the
same functions a React list takes.

### Classes

#### `SortableItem`

One row of a `[molSortableList]`. Its host element is the row dnd-kit makes
focusable and annotates, so the directive needs no template of its own:

```html
<li molSortableItem [molSortableItem]="task.id">{{ task.title }}</li>
```

`exportAs: 'molSortableItem'` exposes `isDragging` and `isDropTarget` to the
template.

```typescript
@Directive({
  selector: '[molSortableItem]',
  standalone: true,
  exportAs: 'molSortableItem',
})
export class SortableItem implements OnInit, OnDestroy {
  /** The id of one of the list's `items`. */
  readonly molSortableItem = input.required<string>()
  /** Data carried with the drag, reported on the drag events. */
  readonly data = input<Record<string, unknown>>()
  /** Disables dragging this item. Defaults to `false`. */
  readonly disabled = input(false)
  /** `true` while this row is being dragged. */
  readonly isDragging = signal(false)
  /** `true` while another dragged row is over this one. */
  readonly isDropTarget = signal(false)

  private readonly list = inject(SORTABLE_LIST, { optional: true })
  private readonly element = inject<ElementRef<Element>>(ElementRef)
  private controller?: SortableItemController

  /**
   * Keeps the row's index, disabled flag and data in step with the list as its
   * `items` change and the list is disabled or enabled.
   */
  constructor() {
    effect(() => {
      const id = this.molSortableItem()
      const disabled = this.disabled()
      const listDisabled = this.list?.disabled() ?? false
      const index = this.list?.indexOf(id) ?? -1
      const data = this.data()
      const controller = this.controller
      if (!controller) return
      controller.setId(id)
      controller.setIndex(index)
      controller.setDisabled(disabled || listDisabled)
      controller.setData(data)
    })
  }

  /**
   * Registers the row with the list's manager, on the host element.
   *
   * @throws {Error} When the row is not inside a `[molSortableList]`.
   */
  ngOnInit(): void {
    const list = this.list
    if (!list) throw new Error('molSortableItem must be used inside a [molSortableList].')
    this.controller = new SortableItemController({
      manager: list.manager,
      list,
      options: {
        id: this.molSortableItem(),
        data: this.data(),
        disabled: this.disabled() || list.disabled(),
      },
      element: this.element.nativeElement,
      isDragging: this.isDragging,
      isDropTarget: this.isDropTarget,
    })
  }

  /**
   * Attaches or clears the element that starts a drag for this row.
   *
   * @param handle - The handle's element, or `undefined` to drag by the whole row.
   */
  setHandle(handle: Element | undefined): void {
    this.controller?.setHandle(handle)
  }

  /** Unregisters the row from the list's manager. */
  ngOnDestroy(): void {
    this.controller?.destroy()
  }
}
```

#### `SortableItemHandle`

Marks an element inside a `[molSortableItem]` as its drag handle: only that
element starts a drag, while the rest of the row stays inert.

```typescript
@Directive({
  selector: '[molSortableItemHandle]',
  standalone: true,
})
export class SortableItemHandle implements OnInit, OnDestroy {
  private readonly item = inject(SortableItem, { optional: true })
  private readonly element = inject<ElementRef<Element>>(ElementRef)

  /**
   * Attaches this element as the row's drag handle.
   *
   * @throws {Error} When the handle is not inside a `[molSortableItem]`.
   */
  ngOnInit(): void {
    if (!this.item)
      throw new Error('molSortableItemHandle must be used inside a [molSortableItem].')
    this.item.setHandle(this.element.nativeElement)
  }

  /** Detaches the handle from the row. */
  ngOnDestroy(): void {
    this.item?.setHandle(undefined)
  }
}
```

#### `SortableList`

A list the user reorders with the pointer, touch or the keyboard (Space or
Enter to pick an item up, the arrow keys to move it, Space or Enter to drop
it, Escape to cancel). A screen reader hears translated instructions and
where the item is as it moves. A drop that moves an item calls `onReorder`
with the new order; render `items` from your state.

The callbacks are inputs, not outputs, so a consumer passes the same
functions a React list takes (`[onReorder]="save"`). Each row carries
{@link SortableItem} (`[molSortableItem]`, or `useSortableItem` in a
component's field initializer).

```typescript
@Directive({
  selector: '[molSortableList]',
  standalone: true,
  providers: [{ provide: SORTABLE_LIST, useExisting: forwardRef(() => SortableList) }],
})
export class SortableList<T extends { id: string } = { id: string }>
  implements SortableListHost, OnDestroy
{
  /** The items, in their current order. */
  readonly items = input<T[]>([])
  /**
   * Called with the whole list in its new order when a drop moves an item. Not
   * called when the drag is cancelled, the list is disabled, or the item is
   * dropped where it started. Persist the order here; the list renders `items`.
   */
  readonly onReorder = input.required<(items: T[]) => void>()
  /** The axis an item may move along. Defaults to `strategy`'s axis, else `'vertical'`. */
  readonly axis = input<SortableAxis>()
  /** The layout: `'grid'` lets items move in any direction. Used when `axis` is absent. */
  readonly strategy = input<SortableStrategy>()
  /** Disables dragging every item. Defaults to `false`. */
  readonly disabled = input(false)
  /** Pointer activation thresholds and the Escape behavior. */
  readonly config = input<DndKitConfig>()
  /**
   * The name a screen reader hears for an item in the drag announcements
   * ("Picked up {label}, position 2 of 5."). Defaults to the item's `id`.
   */
  readonly getItemLabel = input<(item: T) => string>()
  /** Called when a drag starts. */
  readonly onDragStart = input<(event: DragStartEvent) => void>()
  /** Called while a dragged item is over another item. */
  readonly onDragOver = input<(event: DragOverEvent) => void>()
  /** Called when a drag ends, before `onReorder`. `overId` is the item whose place it took. */
  readonly onDragEnd = input<(event: DragEndEvent) => void>()

  /** The dnd-kit manager this list owns. */
  readonly manager: DragDropManager

  /**
   * Creates the list's manager with the shared sensors, modifiers and
   * translated plugins, and reports the manager's drag events to the inputs.
   */
  constructor() {
    this.manager = new DragDropManager({
      plugins: sortableListPlugins({
        labelOf: (id) => {
          const item = this.items().find((candidate) => candidate.id === id)
          const getItemLabel = this.getItemLabel()
          return item && getItemLabel ? getItemLabel(item) : id
        },
        count: () => this.items().length,
      }),
    })
    this.manager.monitor.addEventListener('dragstart', ({ operation }) => {
      const started = coreDragStart(operation.source)
      if (started) this.onDragStart()?.(started)
    })
    this.manager.monitor.addEventListener('dragover', ({ operation }) => {
      const over = coreDragOver(operation.source, operation.target)
      if (over) this.onDragOver()?.(over)
    })
    this.manager.monitor.addEventListener('dragend', (event) => {
      const drop = sortableListDrop(this.items(), event, this.disabled())
      if (drop.event) this.onDragEnd()?.(drop.event)
      if (drop.reordered) this.onReorder()(drop.reordered)
    })
    // Inputs are bound after the constructor, so the sensors and modifiers are
    // rebuilt here, once their inputs are available, the way React's memos do.
    effect(() => {
      const { activationDelay, activationDistance, cancelOnEscape } = this.config() ?? {}
      const sensors = sortableListSensors({
        activationDelay,
        activationDistance,
        cancelOnEscape,
      })
      // The shared helper is typed as a list or a function of the defaults; it returns a list.
      if (Array.isArray(sensors)) this.manager.sensors = sensors
    })
    effect(() => {
      const modifiers = sortableListModifiers(resolveSortableAxis(this.axis(), this.strategy()))
      if (Array.isArray(modifiers)) this.manager.modifiers = modifiers
    })
  }

  /**
   * The item's index in the list's current `items`.
   *
   * @param id - The item's id.
   * @returns Its index, or `-1` when the list has no such item.
   */
  indexOf(id: string): number {
    return this.items().findIndex((item) => item.id === id)
  }

  /** Destroys the manager, which removes the instructions and the live region. */
  ngOnDestroy(): void {
    this.manager.destroy()
  }
}
```

### Functions

#### `useSortableItem(options)`

Makes one row of a `[molSortableList]` draggable. Call it in a component's
field initializer (or constructor) inside the list, then attach `ref` to the
row's element:

```typescript
readonly row = useSortableItem({ id: 'a' })
```

The list's current order and disabled flag are read through signals, so the
row follows them as the list changes without being remounted.

```typescript
function useSortableItem(options: SortableItemOptions): SortableItemState
```

- `options` — The row's item id, drag data and disabled flag.

**Returns:** The refs to attach and the row's drag state.
