Angular quiz / multiple-choice question card — the counterpart of

## Quick Start

```ts
import { Component } from '@angular/core'
import { QuizCard } from '@molecule/app-quiz-card-angular'

@Component({
  selector: 'app-question',
  standalone: true,

    <div mol-quiz-card [progress]="'1 / 10'" [correctId]="'b'" [onAnswer]="onAnswer">
      <span slot="question">What is the capital of France?</span>

export class Question {
  options = [
    { id: 'a', label: 'London' },
    { id: 'b', label: 'Paris' },

  }
}
```

## API

### Interfaces

#### `QuizOption`

A single selectable answer option in a quiz question.

`label` is a string (rendered as text) or an embedded template (a node
input cannot travel through an Angular input, so callers pass nodes as
templates where the framework supplies them).

```typescript
interface QuizOption {
  /** The option's identifier, used for selection and as the rendered key. */
  id: string
  /** The option's label: a plain string, or a template rendering the nodes. */
  label: unknown
}
```

#### `QuizCardProps`

Props accepted by {@link QuizCard} — the same props
`@molecule/app-quiz-card-react` accepts. Node props (`question`, `progress`,
`timer`, `explanation`) are projected content in Angular (an input cannot
carry a node), so they are typed `unknown`: pass them as
`[slot=question]` / `[slot=progress]` / `[slot=timer]` /
`[slot=explanation]` content, or as plain strings.

```typescript
interface QuizCardProps {

  correctId?: string
```

### Classes

#### `QuizCard`

QuizCard component.

Single quiz question card with multiple-choice options. The host IS React's
root `<div>` (`div[mol-quiz-card]`), carrying the `Card` classes exactly as
React's `Card` does. Tracks its own submit state and reveals
correct/incorrect when `correctId` is provided.

The node-valued `question`, `progress`, `timer` and `explanation` props
arrive as projected content (`[slot=question]`, `[slot=progress]`,
`[slot=timer]`, `[slot=explanation]`), as embedded templates, or as plain
string inputs. A wrapper React would not render at all (the progress span,
the explanation paragraph) is detached from the DOM, keeping its capture
point for the projected nodes.

```typescript
@Component({
  selector: 'div[mol-quiz-card]',
  standalone: true,
  imports: [Button, NgTemplateOutlet],
  inputs: [
    'question',
    'options',
    'correctId',
    'onAnswer',
    'progress',
    'timer',
    'explanation',
    'className',
  ],
  host: {
    '[class]': 'cardClasses()',
  },
  template: `<div #stack [class]="stackClasses()">
    <header #header [class]="headerClasses()">
      <span #progressEl [class]="progressClasses()">
        @if (isText(progress)) {
          <ng-container>{{ progress }}</ng-container>
        }
        <ng-container [ngTemplateOutlet]="templateOf(progress)"></ng-container
        ><ng-content select="[slot=progress]"></ng-content
      ></span>
      @if (isText(timer)) {
        <ng-container>{{ timer }}</ng-container>
      }
      <ng-container [ngTemplateOutlet]="templateOf(timer)"></ng-container
      ><ng-content select="[slot=timer]"></ng-content>
    </header>
    <h2 [class]="questionClasses()">
      @if (isText(question)) {
        <ng-container>{{ question }}</ng-container>
      }
      <ng-container [ngTemplateOutlet]="templateOf(question)"></ng-container
      ><ng-content select="[slot=question]"></ng-content>
    </h2>
    <div [class]="optionsClasses()">
      @for (option of options; track option.id) {
        <button
          type="button"
          (click)="select(option)"
          [disabled]="submitted()"
          [attr.aria-pressed]="isSelected(option) ? 'true' : 'false'"
          data-mol-id="quiz-card-option"
          [attr.data-correct]="isCorrect(option) ? 'true' : null"
          [attr.data-wrong]="isWrong(option) ? 'true' : null"
          [class]="optionClasses(option)"
          [style]="optionStyle(option)"
        >
          <span>
            @if (isText(option.label)) {
              <ng-container>{{ option.label }}</ng-container>
            } @else {
              <ng-container [ngTemplateOutlet]="templateOf(option.label)"></ng-container>
            }
          </span>
          @if (isCorrect(option)) {
            <span aria-hidden="true">✓</span>
          }
          @if (isWrong(option)) {
            <span aria-hidden="true">×</span>
          }
        </button>
      }
    </div>
    @if (!submitted()) {
      <button
        mol-button
        [variant]="'solid'"
        [color]="'primary'"
        (click)="submit()"
        [disabled]="!selected()"
      >
        {{ submitLabel() }}
      </button>
    }
    <p #explanationEl [class]="explanationClasses()">
      @if (isText(explanation)) {
        <ng-container>{{ explanation }}</ng-container>
      }
      <ng-container [ngTemplateOutlet]="templateOf(explanation)"></ng-container
      ><ng-content select="[slot=explanation]"></ng-content>
    </p>
  </div>`,
})
export class QuizCard implements AfterViewInit, DoCheck {
  question?: unknown
  options: QuizOption[] = []
  correctId?: string
  onAnswer?: (optionId: string, correct?: boolean) => void
  progress?: unknown
  timer?: unknown
  explanation?: unknown
  className?: string

  @ViewChild('stack') private stack?: ElementRef<HTMLDivElement>
  @ViewChild('header') private header?: ElementRef<HTMLElement>
  @ViewChild('progressEl') private progressEl?: ElementRef<HTMLSpanElement>
  @ViewChild('explanationEl') private explanationEl?: ElementRef<HTMLParagraphElement>

  private readonly i18n = useTranslation()

  private readonly selectedState = signal<string | null>(null)

  private readonly submittedState = signal(false)

  private readonly sync = new SlotWrapper(inject(ElementRef), inject(Renderer2))

  private progressProjected = false
  private explanationProjected = false
  private viewReady = false

  cm(): UIClassMap {
    return getClassMap()
  }

  cardClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.card({ variant: 'elevated' }), cm.cardPadding('md'), this.className)
  }

  stackClasses(): string {
    return this.cm().stack(4)
  }

  headerClasses(): string {
    return this.cm().flex({ justify: 'between', align: 'center', gap: 'sm' })
  }

  progressClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.textSize('xs'), cm.fontWeight('semibold'))
  }

  questionClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.textSize('lg'), cm.fontWeight('bold'))
  }

  optionsClasses(): string {
    return this.cm().stack(2)
  }

  explanationClasses(): string {
    return this.cm().textSize('sm')
  }

  optionClasses(option: QuizOption): string {
    const cm = this.cm()
    return cm.cn(
      cm.flex({ align: 'center', gap: 'sm' }),
      cm.sp('px', 3),
      cm.sp('py', 2),
      cm.textSize('sm'),
      cm.cursorPointer,
      cm.touchTargetCompact,
      this.isSelected(option) ? cm.fontWeight('semibold') : cm.fontWeight('medium'),
    )
  }

  optionStyle(option: QuizOption): string {
    const success = 'var(--mol-color-success, #22c55e)'
    const error = 'var(--mol-color-error, #ef4444)'
    const correct = this.isCorrect(option)
    const wrong = this.isWrong(option)
    const border = correct
      ? success
      : wrong
        ? error
        : this.isSelected(option)
          ? 'currentColor'
          : 'var(--mol-color-border, rgba(128,128,128,0.35))'
    const background = correct
      ? `; background: color-mix(in srgb, ${success} 12%, transparent)`
      : wrong
        ? `; background: color-mix(in srgb, ${error} 12%, transparent)`
        : ''
    return `border-radius: 8px; border: 1px solid ${border}${background}`
  }

  isSelected(option: QuizOption): boolean {
    return option.id === this.selectedState()
  }

  isCorrect(option: QuizOption): boolean {
    return this.submittedState() && !!this.correctId && option.id === this.correctId
  }

  isWrong(option: QuizOption): boolean {
    return (
      this.submittedState() &&
      !!this.correctId &&
      this.isSelected(option) &&
      option.id !== this.correctId
    )
  }

  submitted(): boolean {
    return this.submittedState()
  }

  selected(): string | null {
    return this.selectedState()
  }

  isText(value: unknown): value is string {
    return typeof value === 'string'
  }

  templateOf(value: unknown): TemplateRef<unknown> | null {
    return value instanceof TemplateRef ? value : null
  }

  hasProgress(): boolean {
    return (
      (typeof this.progress === 'string' && this.progress.length > 0) ||
      this.templateOf(this.progress) !== null ||
      this.progressProjected
    )
  }

  hasExplanation(): boolean {
    return (
      (typeof this.explanation === 'string' && this.explanation.length > 0) ||
      this.templateOf(this.explanation) !== null ||
      this.explanationProjected
    )
  }

  submitLabel(): string {
    return this.i18n.t('quizCard.submit', {}, { defaultValue: 'Submit answer' })
  }

  select(option: QuizOption): void {
    if (!this.submittedState()) this.selectedState.set(option.id)
  }

  submit(): void {
    const optionId = this.selectedState()
    if (!optionId || this.submittedState()) return
    this.submittedState.set(true)
    this.onAnswer?.(optionId, this.correctId ? optionId === this.correctId : undefined)
  }

  ngAfterViewInit(): void {
    this.progressProjected = hasNodes(this.progressEl?.nativeElement)
    this.explanationProjected = hasNodes(this.explanationEl?.nativeElement)
    this.viewReady = true
    this.syncWrappers()
  }

  ngDoCheck(): void {
    if (this.viewReady) this.syncWrappers()
  }

  private syncWrappers(): void {
    this.sync.sync(this.progressEl?.nativeElement, this.hasProgress(), this.header?.nativeElement)
    this.sync.sync(
      this.explanationEl?.nativeElement,
      this.submittedState() && this.hasExplanation(),
      this.stack?.nativeElement,
    )
  }
}
```

Companion locale bond: `@molecule/app-locales-quiz-card` (`quizCard.submit`).
The card is single-shot: after submit the options lock and the explanation
replaces the button — there is no reset prop, so advance quizzes by
remounting with a fresh `key` per question. Without `correctId` no
correct/incorrect state is revealed and `onAnswer`'s second argument is
`undefined`. Requires the app-angular i18n provider and a wired ClassMap
bond.

The host element is the card surface itself (`div[mol-quiz-card]`), carrying
the `Card` classes React renders. The node-valued `question`, `progress`,
`timer` and `explanation` props are projected content
(`[slot=question]`, `[slot=progress]`, `[slot=timer]`,
`[slot=explanation]`) or plain strings.
