# @molecule/app-chatbot-tester-angular

`@molecule/app-chatbot-tester-angular` — Angular sandbox UI shell for
live-testing conversational AI: optional bot picker, message transcript,
input row, error display.

Stateless about transport. The consumer owns the message array, send
handler, and bot list; the package handles the layout and composition.

Extracted from the ai-chatbot-builder TestChat page; usable for any
"preview your bot live" UI. This is the Angular counterpart of
`@molecule/app-chatbot-tester-react`: the same props (React names) and the
same DOM for the same input, so an app swaps framework by changing only the
package name in its imports.

      this.messagesOf().length === 0 &&

    this.input = next
  }

  /** Sends the trimmed composer text and clears the field. */
  handleSend = async (): Promise<void> => {
    const text = this.input.trim()
    if (!text) return
    this.input = ''
    await this.onSend?.(text)
  }

  handleBotChange(event: Event): void {
    this.onBotChange?.((event.target as HTMLSelectElement).value)
    this.applySelection()
  }

  /** Reconciles the picker, error and label capture points once the view exists. */
  ngAfterViewInit(): void {
    this.reconcile()
  }

  ngAfterViewChecked(): void {
    this.reconcile()
  }

  /** Applies react-dom's controlled-select selection. */
  private applySelection(): void {
    applyReactSelectSelection(this.botSelect?.nativeElement, this.botId ?? '')
  }

  private reconcile(): void {
    this.guard.toggle(this.botBar?.nativeElement, this.botsOf().length > 1)
    this.guard.toggle(
      this.errorRow?.nativeElement,
      this.errorTemplate() !== null ||
        this.hasErrorText() ||
        hasNodes(this.errorSlot?.nativeElement),
    )
    const labelSlot = this.botLabelSlot?.nativeElement
    if (labelSlot && hasNodes(labelSlot)) this.botLabelDefault?.nativeElement.remove()
    this.applySelection()
  }
}
