# @molecule/app-file-dropzone-angular

Angular file dropzone — drag-drop + click-to-browse file picker with
accept / max-size filtering. Pure UI: it emits the chosen `File[]`
and the host app performs the actual upload.

Exports `<FileDropzone>` and its `FileDropzoneProps`. It is the Angular
counterpart of `@molecule/app-file-dropzone-react`: same props and the same
DOM for the same input.

## Quick Start

```ts
import { Component } from '@angular/core'
import { FileDropzone } from '@molecule/app-file-dropzone-angular'

@Component({
  selector: 'app-avatar-upload',
  standalone: true,
  imports: [FileDropzone],
  template: `
    <div
      mol-file-dropzone
      accept="image/*"
      [maxSize]="5 * 1024 * 1024"
      [onFiles]="uploadAvatar"
      [onRejected]="showError"
    ></div>
  `,
})
export class AvatarUpload {
  avatar: File | null = null
  error = ''
  uploadAvatar = (files: File[]): void => {
    this.avatar = files[0] ?? null
  }
  showError = (files: File[]): void => {
    this.error = `${files.length} file(s) are not images under 5 MB`
  }
}
```

Drag-drop + click-to-browse file upload zone. Pure UI — apps handle the
actual upload. Rejects files exceeding `maxSize` or not matching `accept`
(best-effort extension/MIME check) via `onRejected`.

The host IS the root `<div>` (`div[mol-file-dropzone]`), matching React's
root exactly: `role`, `tabindex`, `data-mol-id`, the drag/click/keyboard

```typescript
export class FileDropzone {
  /** Called with the File[] the user selected or dropped. */
  onFiles?: (files: File[]) => void
  /** Accept attribute (e.g. `'image/*'` or `'.pdf,.docx'`). */
  accept?: string
  /** Allow multi-file selection. Defaults to false. */
  multiple?: boolean
  /** Max file size in bytes. Files exceeding are rejected. */
  maxSize?: number
  /** Called with rejected files (wrong type or size). */
  onRejected?: (files: File[]) => void
  /** Disabled state. */
  disabled?: boolean
  /** Extra classes on the dropzone wrapper. */
  className?: string

  /** The hidden file input (opened by a click or Enter/Space on the zone). */
  @ViewChild('fileInput') private fileInput?: ElementRef<HTMLInputElement>
  /** The projected-content capture (its span never renders; `placeContentBefore` stands in its place). */
  @ViewChild('contentSlot', { static: true }) private contentSlot?: ElementRef<HTMLSpanElement>

  /** The i18n handle (locale-reactive), read in an injection context. */
  private readonly i18n = useTranslation()
  /** Whether a drag is over the zone (React's `useState`). */
  private readonly dragging = signal(false)
  /** Whether the default copy renders (React's `children ?? default`). */
  readonly defaultCopyShown = signal(true)

  /** Renders the projected content in the capture's place and withdraws the capture, then shows the default copy only while nothing is projected, as React's `childr… */
  constructor() {
    usePlacement((hasContent) => {
      const slot = this.contentSlot?.nativeElement
      placeContentBefore(slot, slot)
      setElementPresent(slot, false)
      this.defaultCopyShown.set(!(slot ? hasContent(slot) : false))
    })
  }

  /** The drag-over fill, a theme token mixed into transparency. */
  readonly backgroundStyle = 'color-mix(in srgb, var(--mol-color-primary, #3366ff) 6%, transparent)'

  /**
   * The current class map.
   * @returns The UIClassMap.
   */

    return getClassMap()
  }

  /** The wrapper class list (React's `cm.cn(cm.flex({…}), cm.sp('p', 6), cm.textCenter, !disabled ? cm.cursorPointer : undefined, className)`). */
  classes(): string {
    return this.cm().cn(
      this.cm().flex({ direction: 'col', align: 'center', justify: 'center' }),
      this.cm().sp('p', 6),
      this.cm().textCenter,
      !this.disabled ? this.cm().cursorPointer : undefined,
      this.className,
    )
  }

  /** The dashed drop-surface border (React's `border`), thick in the primary colour while a drag is over the zone and muted otherwise. */
  borderStyle(): string {
    return this.dragging()
      ? '2px dashed var(--mol-color-primary, #3366ff)'
      : '2px dashed var(--mol-color-border, rgba(128,128,128,0.35))'
  }

  /** The default "drop here" paragraph's class (React's `cm.cn(cm.textSize('sm'), cm.fontWeight('semibold'))`). */
  dropHereClasses(): string {
    return this.cm().cn(this.cm().textSize('sm'), this.cm().fontWeight('semibold'))
  }

  /** The accept-hint paragraph's class (React's `cm.cn(cm.textSize('xs'), cm.sp('mt', 1))`). */
  acceptsClasses(): string {
    return this.cm().cn(this.cm().textSize('xs'), this.cm().sp('mt', 1))
  }

  /** The default copy's first line (React's `fileDropzone.dropHere`). */
  dropHereText(): string {
    return this.i18n.t(
      'fileDropzone.dropHere',
      {},
      { defaultValue: 'Drop files here or click to browse' },
    )
  }

  /** The accept hint (React's `fileDropzone.accepts`). */
  acceptsText(): string {
    return this.i18n.t(
      'fileDropzone.accepts',
      { accept: this.accept ?? '' },
      { defaultValue: 'Accepts: {{accept}}' },
    )
  }

  /** Filters a file list into kept and rejected files, exactly as React does: a file over `maxSize` is rejected, then a file matching neither the accept extension n… */
  filter(raw: FileList | File[]): { kept: File[]; rejected: File[] } {
    const arr = Array.from(raw)
    const kept: File[] = []
    const rejected: File[] = []
    for (const f of arr) {
      if (this.maxSize && f.size > this.maxSize) {
        rejected.push(f)
        continue
      }
      // cheap accept check
      if (this.accept) {
        const okByExt = this.accept
          .split(',')
          .some(
            (a) =>
              a.trim() && f.name.toLowerCase().endsWith(a.trim().toLowerCase().replace(/^\./, '.')),
          )
        const okByMime = this.accept
          .split(',')
          .some(
            (a) =>
              a.trim() &&
              f.type &&
              (f.type === a.trim() ||
                (a.trim().endsWith('/*') && f.type.startsWith(a.trim().replace('/*', '/')))),
          )
        if (!okByExt && !okByMime && this.accept !== '*') {
          rejected.push(f)
          continue
        }
      }
      kept.push(f)
    }
    return { kept, rejected }
  }

  /** Filters the given file list and dispatches kept/rejected files to the callbacks — only the first kept file unless `multiple` is set. */
  dispatch(list: FileList | File[]): void {
    const { kept, rejected } = this.filter(list)
    if (kept.length > 0) this.onFiles?.(this.multiple ? kept : [kept[0]!])
    if (rejected.length > 0) this.onRejected?.(rejected)
  }

  /** Opens the native file dialog through the hidden input. */
  openDialog(): void {
    this.fileInput?.nativeElement.click()
  }

  /** React's `onDragOver`: prevents the browser's default (which would open the dropped file) and highlights the zone, unless disabled. */
  onDragOver(event: DragEvent): void {
    if (!this.disabled) {
      event.preventDefault()
      this.dragging.set(true)
    }
  }

  /** React's `onDragLeave`: clears the highlight. */
  onDragLeave(): void {
    this.dragging.set(false)
  }

  /** React's `onDrop`: always prevents the browser's default, clears the highlight, and dispatches the dropped files unless disabled. */
  onDrop(event: DragEvent): void {
    event.preventDefault()
    this.dragging.set(false)
    if (!this.disabled && event.dataTransfer?.files) this.dispatch(event.dataTransfer.files)
  }

  /** React's `onClick`: opens the file dialog unless disabled. */
  onClick(): void {
    if (!this.disabled) this.openDialog()
  }

  /** React's `onKeyDown`: `role="button"` gets no native keyboard activation — Enter/Space open the file dialog and prevent their defaults (Space would otherwise sc… */
  onKeyDown(event: KeyboardEvent): void {
    if (this.disabled) return
    if (event.key === 'Enter' || event.key === ' ' || event.key === 'Spacebar') {
      event.preventDefault()
      this.openDialog()
    }
  }

  /** React's hidden input `onChange`: dispatches the picked files. */
  onInputChange(event: Event): void {
    const files = (event.target as HTMLInputElement).files
    if (files) this.dispatch(files)
  }
}
```

With `multiple` false (the default) only the FIRST accepted file is
delivered to `onFiles`, even when the user drops several at once.

The `accept` filter is a best-effort client-side check (file
extension or MIME prefix match) — treat it as UX sugar and always
re-validate on the server. Files failing `accept` or `maxSize` go to
`onRejected` (both rejection causes arrive in the same callback).

`children` is projected content (Angular cannot carry a node in an input):
put it between the element's tags to replace the default "Drop files here or
click to browse" copy, which translates via
`@molecule/app-locales-file-dropzone`.

The zone is a keyboard-activatable `role="button"` (`tabindex="0"`):
pressing Enter or Space opens the native file dialog, matching a click.
