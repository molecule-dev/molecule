# @molecule/app-feature-image-canvas-angular

Angular Canvas2D image-editing surface.

Exports `<div mol-image-canvas>` — the Angular counterpart of
`@molecule/app-feature-image-canvas-react` with the same props and the same
DOM: an image rendered onto a canvas with a filter chain (brightness,
contrast, saturation, hue, sepia, grayscale, blur, sharpen), pointer-drag
panning, wheel zooming, and a `toDataURL()` export handle, plus the
framework-free `clamp`, `composeFilterString`, `screenToCanvas` and

#### `ImageCanvasFilters`

Filter parameters applied to the source image before it is rendered to
the canvas. Every field is optional — omitted fields disable that
filter. Values are clamped at compose time so out-of-range numbers are
safe.

Implementation note: composition uses the Canvas2D `filter` property
(CSS `<filter-function>` syntax). Compose order is preserved.

```typescript
interface ImageCanvasFilters {
  /**
   * Brightness multiplier. `1` is identity, `0` is black, `2` is twice
   * as bright. Values < 0 are clamped to 0.
   */
  brightness?: number
  /**
   * Contrast multiplier. `1` is identity, `0` is fully grey, `2` is
   * twice the contrast. Values < 0 are clamped to 0.
   */
  contrast?: number
  /**
   * Saturation multiplier. `1` is identity, `0` is grayscale, `2` is
   * twice as saturated. Values < 0 are clamped to 0.
   */
  saturation?: number
  /**
   * Hue rotation in degrees. Normalised modulo 360 — any finite value
   * is accepted.
   */
  hue?: number
  /**
   * Sepia amount in `[0, 1]`. `0` disables; `1` applies fully. Values
   * outside the range are clamped.
   */
  sepia?: number
  /**
   * Grayscale amount in `[0, 1]`. `0` disables; `1` applies fully.
   * Values outside the range are clamped.

  grayscale?: number

   * Gaussian blur radius in CSS pixels. `0` disables. Negative values
   * are clamped to 0.
   */
  blur?: number
```

```typescript
  src?: string | HTMLImageElement
  filters?: ImageCanvasFilters
  /** Zoom factor; values <= 0 fall back to `1`. */
  zoom?: number
  pan?: PanOffset
  onChange?: (next: { zoom: number; pan: PanOffset }) => void

  width?: number
  height?: number
  className?: string

  @ViewChild('surface') private surface?: ElementRef<HTMLCanvasElement>

  private readonly i18n = useTranslation()
  private t = this.i18n.t
  private readonly classMap: UIClassMap = getClassMap()

  private readonly image = signal<HTMLImageElement | null>(null)
  private readonly loadError = signal<string | null>(null)
  private loadToken = 0
  private viewReady = false
  private dragState: {
    pointerId: number | null
    startX: number
    startY: number
    startPan: PanOffset
  } = { pointerId: null, startX: 0, startY: 0, startPan: { x: 0, y: 0 } }

  /** The region's translated aria label. */
  protected readonly ariaLabel = computed(() =>

  /** The canvas's translated aria label. */
  protected readonly dragLabel = computed(() =>

  /** The translated loading message. */

  )

  /** The translated error message. */
  protected readonly errorLabel = computed(() =>
    this.t('imageCanvas.error', {}, { defaultValue: 'Image failed to load.' }),
  )

  protected readonly loadingAttr = computed(() =>
    this.image() ? 'false' : this.loadError() ? 'false' : 'true',

  protected readonly errorAttr = computed(() => (this.loadError() ? 'true' : 'false'))

  ngOnInit(): void {
    const src = this.src
    if (typeof src === 'string') {
      this.load(src)
    } else if (src !== undefined && isImageElement(src)) {
      this.image.set(src)
      this.loadError.set(null)
    }
  }

  ngOnChanges(changes: SimpleChanges): void {
    if (!this.viewReady) return
    if (changes['src']) {
      const src = changes['src'].currentValue as string | HTMLImageElement | undefined
      if (typeof src === 'string') this.load(src)
      else if (src !== undefined && isImageElement(src)) {
        this.image.set(src)
        this.loadError.set(null)
      }
    }
    this.paint()
  }

  ngAfterViewInit(): void {
    this.viewReady = true
    this.paint()
    if (this.exportRef) {
      this.exportRef.current = {
        toDataURL: (type, quality) => {
          const canvas = this.surface?.nativeElement
          if (!canvas) return ''
          return canvas.toDataURL(type, quality)
        },
      }
    }
  }

  ngOnDestroy(): void {
    if (this.exportRef) this.exportRef.current = null
  }

  classes(): string {
    return this.classMap.cn(this.className)
  }

  widthValue(): number {
    return this.width ?? 512
  }

  heightValue(): number {
    return this.height ?? 512
  }

  cursor(): string {
    return this.onChange ? 'grab' : 'default'
  }

  /** The loading/error paragraph's class list. */
  noteClasses(): string {
    return this.classMap.cn(this.classMap.textSize('xs'), this.classMap.sp('p', 2))
  }

  handlePointerDown(event: PointerEvent): void {
    const canvas = this.surface?.nativeElement
    if (!canvas) return
    canvas.setPointerCapture(event.pointerId)
    this.dragState = {
      pointerId: event.pointerId,
      startX: event.clientX,
      startY: event.clientY,
      startPan: { ...this.safePan() },
    }
  }

  handlePointerMove(event: PointerEvent): void {
    const state = this.dragState
    if (state.pointerId !== event.pointerId) return
    const dx = event.clientX - state.startX
    const dy = event.clientY - state.startY
    const image = this.image()
    const next = clampPan(
      { x: state.startPan.x + dx, y: state.startPan.y + dy },
      { width: this.widthValue(), height: this.heightValue() },
      image
        ? { width: image.naturalWidth, height: image.naturalHeight }
        : { width: this.widthValue(), height: this.heightValue() },
      this.safeZoom(),
    )
    this.onChange?.({ zoom: this.safeZoom(), pan: next })
  }

  endDrag(event: PointerEvent): void {
    const state = this.dragState
    if (state.pointerId !== event.pointerId) return
    const canvas = this.surface?.nativeElement
    if (canvas && canvas.hasPointerCapture(event.pointerId)) {
      canvas.releasePointerCapture(event.pointerId)
    }
    this.dragState = { pointerId: null, startX: 0, startY: 0, startPan: { x: 0, y: 0 } }
  }

  handleWheel(event: WheelEvent): void {
    const factor = event.deltaY < 0 ? 1.1 : 1 / 1.1
    const nextZoom = clamp(this.safeZoom() * factor, 0.05, 32)
    this.onChange?.({ zoom: nextZoom, pan: this.safePan() })
  }

  private safeZoom(): number {
    const zoom = this.zoom
    return zoom !== undefined && Number.isFinite(zoom) && zoom > 0 ? zoom : 1
  }

  private safePan(): PanOffset {
    return {
      x: Number.isFinite(this.pan?.x) ? (this.pan?.x as number) : 0,
      y: Number.isFinite(this.pan?.y) ? (this.pan?.y as number) : 0,
    }
  }

  private load(src: string): void {
    const token = ++this.loadToken
    this.loadError.set(null)
    const img = new Image()
    img.crossOrigin = 'anonymous'
    img.onload = () => {
      if (token !== this.loadToken) return
      this.image.set(img)
      this.paint()
    }
    img.onerror = () => {
      if (token !== this.loadToken) return
      this.loadError.set(src)
      this.image.set(null)
      this.paint()
    }
    img.src = src
  }

  private paint(): void {
    const canvas = this.surface?.nativeElement
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    if (!ctx) return

    const image = this.image()
    const zoom = this.safeZoom()
    const pan = this.safePan()
    const width = this.widthValue()
    const height = this.heightValue()

    ctx.save()
    ctx.clearRect(0, 0, width, height)

    if (!image) {
      ctx.restore()
      return
    }

    ctx.filter = composeFilterString(this.filters)
    const drawWidth = image.naturalWidth * zoom
    const drawHeight = image.naturalHeight * zoom
    const dx = width / 2 - drawWidth / 2 + pan.x
    const dy = height / 2 - drawHeight / 2 + pan.y
    ctx.drawImage(image, dx, dy, drawWidth, drawHeight)
    ctx.restore()
  }
}
```

### Functions

#### `clamp(value, min, max)`

Clamp a numeric value into the closed interval `[min, max]`. Returns
`min` for non-finite inputs.

```typescript
function clamp(value: number, min: number, max: number): number
```

- `value` — The value to clamp.
- `min` — Inclusive minimum.
- `max` — Inclusive maximum.

**Returns:** The clamped value.

#### `composeFilterString(filters)`

Compose a CSS filter string from a set of `ImageCanvasFilters` values.
Out-of-range values are clamped; omitted / non-finite fields are
ignored. Returns `'none'` when no filters are active so callers can
assign the result directly to `CanvasRenderingContext2D.filter`.

```typescript
function composeFilterString(filters?: ImageCanvasFilters): string
```

- `filters` — The filter parameters. May be `undefined`.

**Returns:** A CSS filter string, or `'none'` when nothing applies.

#### `screenToCanvas(point, canvasSize, imageSize, zoom, pan)`

Convert a screen-space (canvas-relative) point into image-space
coordinates given the current zoom + pan and the source image
dimensions. Useful for tools that need to know which image pixel a
pointer event hit (cropping, picking, masking).

```typescript
function screenToCanvas(
  point: { x: number; y: number },
  canvasSize: { width: number; height: number },
  imageSize: { width: number; height: number },
  zoom: number,
  pan: PanOffset,
): { x: number; y: number }
```

- `point` — The canvas-space point in CSS pixels (origin top-left of the canvas element).
- `canvasSize` — The canvas size in CSS pixels.
- `imageSize` — The source image's natural size in pixels.
- `zoom` — The current zoom factor. Non-positive values fall back to `1`.
- `pan` — The current pan offset in CSS pixels.

**Returns:** The corresponding point in image-space coordinates.

#### `clampPan(pan, canvasSize, imageSize, zoom)`

Clamp a pan offset so the (zoomed) image cannot be dragged entirely
off-canvas. The clamp keeps at least half of the image visible in
each axis. Non-finite inputs are treated as `0`.

```typescript
function clampPan(
  pan: PanOffset,
  canvasSize: { width: number; height: number },
  imageSize: { width: number; height: number },
  zoom: number,
): PanOffset
```

- `pan` — The desired pan offset, in CSS pixels.
- `canvasSize` — The canvas size in CSS pixels.
- `imageSize` — The source image's natural size in pixels.
- `zoom` — The current zoom factor. Non-positive values fall back to `1`.

**Returns:** The clamped pan offset.
