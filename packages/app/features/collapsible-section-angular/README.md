### Classes

#### `CollapsibleSection`

CollapsibleSection component.

Expandable section with a clickable heading. Lighter-weight than an
accordion — renders a single toggle, not a group of panels. The host IS the
root `<section>` (`section[mol-collapsible-section]`), matching React's root
exactly.

Node props are projected content (`[slot=title]`, `[slot=badge]`,
`[slot=actions]`, and the default projection for the body) or template refs
on the same-named inputs, as in `@molecule/app-announcement-bar-angular`.
React renders the body `<div>` only while expanded and the actions `<span>`
only when `actions` is given; Angular projects into always-mounted capture
points, so {@link CollapsibleSection.ngAfterViewChecked} detaches and restores those
wrappers to match.

```typescript
@Component({
  selector: 'section[mol-collapsible-section]',
  standalone: true,
  imports: [NgTemplateOutlet],
  inputs: [
    'title',
    'badge',
    'actions',
    'expanded',
    'defaultExpanded',
    'onExpandedChange',
    'level',
    'className',
  ],
  host: {
    '[class]': 'classes()',
  },
  template: `@switch (levelTag()) {
      @case (2) {
        <h2><ng-container [ngTemplateOutlet]="headingBody" /></h2>
      }
      @case (4) {
        <h4><ng-container [ngTemplateOutlet]="headingBody" /></h4>
      }
      @case (5) {
        <h5><ng-container [ngTemplateOutlet]="headingBody" /></h5>
      }
      @case (6) {
        <h6><ng-container [ngTemplateOutlet]="headingBody" /></h6>
      }
      @default {
        <h3><ng-container [ngTemplateOutlet]="headingBody" /></h3>
      }
    }
    <div #bodyWrap><ng-content></ng-content></div>
    <ng-template #headingBody>
      <!-- The header row is a container, not a button: a button may not
           contain other controls, and "actions" typically holds them. The
           toggle wraps the label; "actions" sits beside it, so a click there
           never reaches the toggle (no stopPropagation needed). -->
      <span [class]="headerClasses()">
        <button
          type="button"
          (click)="toggle()"
          [attr.aria-expanded]="isOpen() ? 'true' : 'false'"
          [class]="buttonClasses()"
        >
          <span aria-hidden="true">{{ isOpen() ? '▾' : '▸' }}</span>
          <span
            ><ng-content select="[slot=title]"></ng-content>
            @if (titleTemplate(); as titleTpl) {
              <ng-container [ngTemplateOutlet]="titleTpl" />
            } @else if (titleText()) {
              {{ title }}
            }
          </span>
          <ng-content select="[slot=badge]"></ng-content>
          @if (badgeTemplate(); as badgeTpl) {
            <ng-container [ngTemplateOutlet]="badgeTpl" />
          } @else if (badgeText()) {
            {{ badge }}
          }
        </button>
        <span #actionsWrap
          ><ng-content select="[slot=actions]"></ng-content>
          @if (actionTemplate(); as actionTpl) {
            <ng-container [ngTemplateOutlet]="actionTpl" />
          } @else if (actionText()) {
            {{ actions }}
          }
        </span>
      </span>
    </ng-template>`,
})
export class CollapsibleSection implements OnInit, AfterViewInit, AfterViewChecked {
  title?: TemplateRef<unknown> | string | null
  badge?: TemplateRef<unknown> | string | null
  actions?: TemplateRef<unknown> | string | null
  expanded?: boolean
  defaultExpanded?: boolean
  onExpandedChange?: (expanded: boolean) => void
  level?: 2 | 3 | 4 | 5 | 6
  className?: string

  @ViewChild('bodyWrap') bodyWrap?: ElementRef<HTMLElement>
  @ViewChild('actionsWrap') actionsWrap?: ElementRef<HTMLElement>

  private readonly internal = signal(false)
  private readonly bodyState: AttachedState = { parent: null }
  private readonly actionsState: AttachedState = { parent: null }
  private actionsRead = false
  private hasActionsSlot = false

  ngOnInit(): void {
    this.internal.set(this.defaultExpanded ?? false)
  }

  cm(): UIClassMap {
    return getClassMap()
  }

  classes(): string {
    return this.cm().cn(this.cm().stack(2), this.className)
  }

  headerClasses(): string {
    return this.cm().cn(
      this.cm().flex({ align: 'center', justify: 'between', gap: 'sm' }),
      this.cm().w('full'),
      this.cm().sp('py', 2),
      this.cm().textSize('sm'),
      this.cm().fontWeight('semibold'),
    )
  }

  buttonClasses(): string {
    return this.cm().flex({ align: 'center', gap: 'sm' })
  }

  isOpen(): boolean {
    return this.expanded ?? this.internal()
  }

  levelTag(): number {
    const level = this.level
    return level === 2 || level === 4 || level === 5 || level === 6 ? level : 3
  }

  toggle(): void {
    const next = !this.isOpen()
    if (this.expanded === undefined) this.internal.set(next)
    this.onExpandedChange?.(next)
  }

  titleTemplate(): TemplateRef<unknown> | null {
    return isTemplate(this.title) ? this.title : null
  }

  titleText(): string {
    return typeof this.title === 'string' || typeof this.title === 'number'
      ? textOf(this.title)
      : ''
  }

  badgeTemplate(): TemplateRef<unknown> | null {
    return isTemplate(this.badge) ? this.badge : null
  }

  badgeText(): string {
    return typeof this.badge === 'string' || typeof this.badge === 'number'
      ? textOf(this.badge)
      : ''
  }

  actionTemplate(): TemplateRef<unknown> | null {
    return isTemplate(this.actions) ? this.actions : null
  }

  actionText(): string {
    return typeof this.actions === 'string' || typeof this.actions === 'number'
      ? textOf(this.actions)
      : ''
  }

  actionsContent(): boolean {
    if (this.actions !== undefined && this.actions !== null) return Boolean(this.actions)
    return this.hasActionsSlot
  }

  ngAfterViewInit(): void {
    this.reconcile()
  }

  ngAfterViewChecked(): void {
    this.reconcile()
  }

  private reconcile(): void {
    const actions = this.actionsWrap?.nativeElement
    if (!this.actionsRead && actions) {
      this.hasActionsSlot = hasNodes(actions)
      this.actionsRead = true
    }
    setAttached(actions, this.actionsContent(), this.actionsState)
    setAttached(this.bodyWrap?.nativeElement, this.isOpen(), this.bodyState)
  }
}
```

#### `ShowMore`

ShowMore component.

"Show N more" / "Show less" toggle: React truncates its `children` array
with `children.slice(0, initialCount)`. Angular projects the items into the
host (`div[mol-show-more]`, matching React's root exactly) and a projected
list has no `length`, so {@link ShowMore.ngAfterViewInit} measures the
projected nodes once and {@link ShowMore.ngAfterViewChecked} detaches the ones past
`initialCount`, restoring them in order when expanded. The labels come from
`useTranslation()` (`showMore.more` / `showMore.less`), read reactively so a
locale switch refreshes them.

```typescript
@Component({
  selector: 'div[mol-show-more]',
  standalone: true,
  inputs: ['initialCount', 'moreKey', 'lessKey', 'className'],
  host: {
    '[class]': 'classes()',
  },
  template: `<ng-content></ng-content>
    @if (remaining() > 0) {
      <button #toggleBtn type="button" (click)="toggle()" [class]="buttonClasses()">
        {{ label() }}
      </button>
    }`,
})
export class ShowMore implements AfterViewInit, AfterViewChecked {
  /** Number to show initially (defaults to 3 at read time). */
  initialCount?: number
  /** i18n key for the "Show more" label (defaults to `showMore.more`). */
  moreKey?: string
  /** i18n key for the "Show less" label (defaults to `showMore.less`). */
  lessKey?: string
  /** Extra classes. */
  className?: string

  /** The toggle button, while React renders one. */
  @ViewChild('toggleBtn') toggleBtn?: ElementRef<HTMLElement>

  /** The host element (the root `<div>`), which holds the projected items. */
  private readonly host = inject(ElementRef) as ElementRef<HTMLElement>
  /** Reactive translation, read in the injection context. */
  private readonly i18n = useTranslation()
  /** Whether every item shows (React's `expanded`). */
  private readonly expanded = signal(false)
  /** How many items were projected (-1 until measured). */
  private readonly count = signal(-1)
  /** The projected item nodes, in document order (captured once). */
  private items: (Element | Text)[] = []

  /**
   * The current class map.
   * @returns The UIClassMap.
   */
  cm(): UIClassMap {
    return getClassMap()
  }

  /**
   * Computes the root div's class list (React's `cm.cn(cm.stack(2), className)`).
   * @returns The class string.
   */
  classes(): string {
    return this.cm().cn(this.cm().stack(2), this.className)
  }

  /**
   * Computes the toggle button's class list.
   * @returns The class string.
   */
  buttonClasses(): string {
    return this.cm().cn(this.cm().textSize('sm'), this.cm().fontWeight('semibold'), this.cm().link)
  }

  /**
   * React's `initialCount = 3` default parameter.
   * @returns The initial count.
   */
  initial(): number {
    return this.initialCount ?? 3
  }

  /**
   * How many items the toggle reveals (React's `Math.max(0, children.length - initialCount)`).
   * @returns The remaining count; 0 until the projected items are measured.
   */
  remaining(): number {
    return this.count() < 0 ? 0 : Math.max(0, this.count() - this.initial())
  }

  /**
   * The toggle's label — the "Show N more" text, or "Show less" when expanded.
   * @returns The label text.
   */
  label(): string {
    if (this.expanded())
      return this.i18n.t(this.lessKey ?? 'showMore.less', {}, { defaultValue: 'Show less' })
    const remaining = this.remaining()
    return this.i18n.t(
      this.moreKey ?? 'showMore.more',
      { remaining },
      { defaultValue: `Show ${remaining} more` },
    )
  }

  /** Flips the expanded state (React's `setExpanded((x) => !x)`). */
  toggle(): void {
    this.expanded.set(!this.expanded())
  }

  /**
   * Measures the projected items (view queries and projection resolve only at
   * view init) and reconciles the DOM with React's slice.
   */
  ngAfterViewInit(): void {
    this.items = itemNodes(this.host.nativeElement)
    this.count.set(this.items.length)
    this.reconcile()
  }

  /**
   * Re-reconciles after the view is checked: an input write (a props step may
   * change `initialCount`) or the toggle button being created, which gives the
   * restored items their anchor.
   */
  ngAfterViewChecked(): void {
    this.reconcile()
  }

  /**
   * Shows the first `initialCount` items (all of them when expanded) and hides
   * the rest, as React's `children.slice(0, initialCount)` renders.
   */
  private reconcile(): void {
    if (this.items.length === 0) return
    const anchor = this.toggleBtn?.nativeElement ?? null
    truncateItems(
      this.host.nativeElement,
      this.items,
      this.expanded() ? this.items.length : this.initial(),
      anchor,
    )
  }
}
```

Node props are projected content: `[slot=title]`, `[slot=badge]` and
`[slot=actions]` fill the heading, and the element's other content is the
body (or, for `<ShowMore>`, the list items). The same values also work as
template refs on the `title` / `badge` / `actions` inputs.
`<CollapsibleSection>` is uncontrolled by default (`defaultExpanded`); pass
`expanded` to control it — then update it from `onExpandedChange`. The header
renders as a single `<button>`, so `[slot=actions]` content must NOT contain
buttons/links (nested interactive elements are invalid HTML).
`<ShowMore>`'s labels use the i18n keys `showMore.more` / `showMore.less`
with English `defaultValue`s; no companion locale bond ships these keys —
add them to your app's locale resources (or pass custom
`moreKey`/`lessKey`). A projected list has no `length`, so `<ShowMore>`
measures the projected nodes to decide which ones to hide — project the
items themselves, not a wrapper around them.
