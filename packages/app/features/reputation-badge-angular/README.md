# @molecule/app-reputation-badge-angular

Angular reputation/karma surfaces — `<ReputationBadge>` and `<BadgeShelf>` —
for forum, discussion-board, and social flagships.

`<ReputationBadge score={n} />` renders a compact-formatted score ("1.5k")
plus a tier chip; the tier derives from `score` via `levelForScore` and
`DEFAULT_THRESHOLDS` (contributor 100, trusted 500, veteran 2000,
legend 10000) unless you pass `level` or `thresholds`. `variant` is
`'compact'` (default) or `'full'` (stacked with a "Reputation" caption).
`<BadgeShelf badges limit onClick />` renders earned-badge icons with a
trailing `+N` overflow chip (`onClick(null)` signals "expand all").

This is the Angular counterpart of `@molecule/app-reputation-badge-react`:
same exports, same props (React names), same DOM for the same input.

#### `ReputationThresholds`

Optional thresholds for deriving a `ReputationLevel` from a numeric score.

If a `level` prop isn't passed to `<ReputationBadge>`, the component will
pick the highest level whose threshold the `score` meets or exceeds.

Default thresholds:

- `contributor` >= 100
- `trusted` >= 500
- `veteran` >= 2000
- `legend` >= 10000
- below 100 → `newcomer`

### Classes

#### `BadgeShelf`

BadgeShelf component.

Horizontal row of small badge icons earned by a user. The host is a wrapper
the DOM normalizer unwraps (`data-parity-unwrap`): React's root is the `<ul>`
itself, and renders nothing at all when no badges are supplied, so the `<ul>`
exists only while there is something to show.

Each icon shows its name in a native `title` tooltip. Clicking a badge
invokes `onClick(badge)`; clicking the trailing `+N` chip invokes
`onClick(null)`.

```typescript
@Component({
  selector: 'mol-badge-shelf',
  standalone: true,
  inputs: ['badges', 'limit', 'onClick', 'className'],
  host: { 'data-parity-unwrap': '', style: 'display: contents' },
  template: `@if (hasBadges()) {
    <ul [class]="rootClasses()" [attr.aria-label]="shelfLabel()" data-mol-id="badge-shelf">
      @for (badge of visible(); track badge.id) {
        <li [class]="itemClasses()" [attr.data-badge-id]="badge.id" data-mol-id="badge-shelf-item">
          @if (isInteractive()) {
            <button
              type="button"
              [class]="buttonClasses()"
              (click)="invoke(badge)"
              [attr.title]="tooltip(badge)"
              [attr.aria-label]="badge.name"
              data-mol-id="badge-shelf-button"
            >
              @if (badge.iconSrc) {
                <img
                  [src]="badge.iconSrc"
                  alt=""
                  [class]="iconClasses()"
                  style="width: 24px; height: 24px"
                  data-mol-id="badge-shelf-icon-img"
                />
              } @else {
                <span aria-hidden="true" data-mol-id="badge-shelf-icon-glyph">{{
                  badge.icon ?? '*'
                }}</span>
              }
            </button>
          } @else {
            <span [attr.title]="tooltip(badge)" [attr.aria-label]="badge.name" role="img">
              @if (badge.iconSrc) {
                <img
                  [src]="badge.iconSrc"
                  alt=""
                  [class]="iconClasses()"
                  style="width: 24px; height: 24px"
                  data-mol-id="badge-shelf-icon-img"
                />
              } @else {
                <span aria-hidden="true" data-mol-id="badge-shelf-icon-glyph">{{
                  badge.icon ?? '*'
                }}</span>
              }
            </span>
          }
        </li>
      }
      @if (overflow() > 0) {
        <li [class]="itemClasses()" data-mol-id="badge-shelf-overflow">
          @if (isInteractive()) {
            <button
              type="button"
              [class]="overflowButtonClasses()"
              (click)="invokeOverflow()"
              [attr.aria-label]="overflowLabel()"
              data-mol-id="badge-shelf-overflow-button"
            >
              +{{ overflow() }}
            </button>
          } @else {
            <span [class]="overflowTextClasses()" [attr.aria-label]="overflowLabel()"
              >+{{ overflow() }}</span
            >
          }
        </li>
      }
    </ul>
  }`,
})
export class BadgeShelf {
  /** Earned badges to display, ordered most-prominent-first. */
  badges!: Badge[]
  /** Maximum number of badge icons rendered inline (React's default: 5). */
  limit?: number
  /** Click handler, called with the badge (or `null` for the overflow chip). */
  onClick?: (badge: Badge | null) => void
  /** Extra classes merged onto the outer row. */
  className?: string

  /** The i18n handle (locale-reactive), read in an injection context. */
  private readonly i18n = useTranslation()

  /**
   * The current class map.
   * @returns The UIClassMap.
   */
  cm(): UIClassMap {
    return getClassMap()
  }

  /**
   * Whether React renders the `<ul>` at all (`badges.length > 0`).
   * @returns True when at least one badge was supplied.
   */
  hasBadges(): boolean {
    return (this.badges?.length ?? 0) > 0
  }

  /**
   * The badges React renders inline (the first `limit`, default 5).
   * @returns The visible badges.
   */
  visible(): Badge[] {
    return (this.badges ?? []).slice(0, this.limit ?? 5)
  }

  /**
   * The number of badges summarised by the trailing `+N` chip.
   * @returns The overflow count (never negative).
   */
  overflow(): number {
    return Math.max(0, (this.badges?.length ?? 0) - this.visible().length)
  }

  /**
   * Whether React makes the icons clickable (`typeof onClick === 'function'`).
   * @returns True when a click handler was given.
   */
  isInteractive(): boolean {
    return typeof this.onClick === 'function'
  }

  /**
   * The badge's native tooltip: its name, plus its description when set.
   * @param badge - The badge.
   * @returns The tooltip text.
   */
  tooltip(badge: Badge): string {
    return badge.description ? `${badge.name} — ${badge.description}` : badge.name
  }

  /**
   * The row's accessible label.
   * @returns The `aria-label`.
   */
  shelfLabel(): string {
    return this.i18n.t(
      'badgeShelf.aria',
      { count: this.badges?.length ?? 0 },
      { defaultValue: '{{count}} badges earned' },
    )
  }

  /**
   * The overflow chip's accessible label.
   * @returns The `aria-label`.
   */
  overflowLabel(): string {
    return this.i18n.t(
      'badgeShelf.overflow.aria',
      { count: this.overflow() },
      { defaultValue: 'Show {{count}} more badges' },
    )
  }

  /**
   * The `<ul>`'s class list.
   * @returns The class string.
   */
  rootClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.flex({ align: 'center', gap: 'xs' }), this.className)
  }

  /**
   * One `<li>`'s class list.
   * @returns The class string.
   */
  itemClasses(): string {
    return this.cm().cn(this.cm().flex({ align: 'center', justify: 'center' }))
  }

  /**
   * An interactive badge button's class list.
   * @returns The class string.
   */
  buttonClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.flex({ align: 'center', justify: 'center' }), cm.roundedFull)
  }

  /**
   * The overflow button's class list.
   * @returns The class string.
   */
  overflowButtonClasses(): string {
    const cm = this.cm()
    return cm.cn(
      cm.flex({ align: 'center', justify: 'center' }),
      cm.roundedFull,
      cm.textSize('xs'),
      cm.fontWeight('semibold'),
    )
  }

  /**
   * The non-interactive overflow chip's class list.
   * @returns The class string.
   */
  overflowTextClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.textSize('xs'), cm.fontWeight('semibold'))
  }

  /**
   * An image icon's class list.
   * @returns The class string.
   */
  iconClasses(): string {
    return this.cm().cn(this.cm().shrink0)
  }

  /**
   * Calls the click handler with the clicked badge.
   * @param badge - The badge whose icon was clicked.
   */
  invoke(badge: Badge): void {
    this.onClick?.(badge)
  }

  /** Calls the click handler with `null`, React's "expand all" signal. */
  invokeOverflow(): void {
    this.onClick?.(null)
  }
}
```

#### `ReputationBadge`

ReputationBadge component.

User reputation / karma display: a numeric score paired with a tier chip
whose color reflects the derived (or explicit) level. The host is a wrapper
the DOM normalizer unwraps (`data-parity-unwrap`) because React's root
element flips between `<span>` (compact) and `<div>` (full); the rendered
root is the only child of the host, exactly React's root.

Tier color comes from the active ClassMap bond's `cm.badge()` helper, so apps
swapping styling libraries restyle these chips for free.

```typescript
@Component({
  selector: 'mol-reputation-badge',
  standalone: true,
  imports: [Badge],
  inputs: ['score', 'level', 'thresholds', 'variant', 'className'],
  host: { 'data-parity-unwrap': '', style: 'display: contents' },
  template: `@if (isFull()) {
      <div [class]="fullClasses()" [attr.aria-label]="ariaLabel()" data-mol-id="reputation-badge">
        <span [class]="captionClasses()">{{ caption() }}</span>
        <div [class]="rowClasses()" data-mol-id="reputation-badge-row">
          <span [class]="fullScoreClasses()" data-mol-id="reputation-badge-score">{{
            formattedScore()
          }}</span>
          <span mol-badge [color]="tierColor()">{{ levelLabelText() }}</span>
        </div>
      </div>
    } @else {
      <span
        [class]="compactClasses()"
        [attr.aria-label]="ariaLabel()"
        data-mol-id="reputation-badge"
      >
        <span [class]="compactScoreClasses()" data-mol-id="reputation-badge-score">{{
          formattedScore()
        }}</span>
        <span mol-badge [color]="tierColor()">{{ levelLabelText() }}</span>
      </span>
    }`,
})
export class ReputationBadge {
  /** Numeric reputation / karma score. */
  score!: number
  /** Explicit reputation level (derived from `score` when omitted). */
  level?: ReputationLevel
  /** Override the default score → level thresholds. */
  thresholds?: ReputationThresholds
  /** Layout variant: `'compact'` (default) or `'full'`. */
  variant?: 'compact' | 'full'
  /** Extra classes merged onto the outer wrapper. */
  className?: string

  /** The i18n handle (locale-reactive), read in an injection context. */
  private readonly i18n = useTranslation()

  /**
   * Whether React renders the full (stacked) variant.
   * @returns True only for `variant === 'full'`.
   */
  isFull(): boolean {
    return this.variant === 'full'
  }

  /**
   * The current class map.
   * @returns The UIClassMap.
   */
  cm(): UIClassMap {
    return getClassMap()
  }

  /**
   * The resolved reputation level (React's `level ?? levelForScore(...)`).
   * @returns The level.
   */
  resolvedLevel(): ReputationLevel {
    return this.level ?? levelForScore(this.score, this.thresholds ?? DEFAULT_THRESHOLDS)
  }

  /**
   * The tier chip's semantic color.
   * @returns The Badge color.
   */
  tierColor(): ReturnType<typeof colorForLevel> {
    return colorForLevel(this.resolvedLevel())
  }

  /**
   * The translated level label — exactly React's `levelLabel`: one LITERAL i18n
   * key per in-contract level, and the literal `reputationBadge.level.fallback`
   * key (whose English is the bare `{{level}}` placeholder, so an unknown id
   * renders as itself) for anything else.
   *
   * Literal keys, never one built from the level id: the i18n gate can only see
   * literal keys, so a key built at runtime that no locale bond carries leaves
   * every non-English user on English with nothing reporting it.
   * @returns The label text.
   */
  levelLabelText(): string {
    switch (this.resolvedLevel()) {
      case 'newcomer':
        return this.i18n.t('reputationBadge.level.newcomer', {}, { defaultValue: 'Newcomer' })
      case 'contributor':
        return this.i18n.t('reputationBadge.level.contributor', {}, { defaultValue: 'Contributor' })
      case 'trusted':
        return this.i18n.t('reputationBadge.level.trusted', {}, { defaultValue: 'Trusted' })
      case 'veteran':
        return this.i18n.t('reputationBadge.level.veteran', {}, { defaultValue: 'Veteran' })
      case 'legend':
        return this.i18n.t('reputationBadge.level.legend', {}, { defaultValue: 'Legend' })
      default:
        return this.i18n.t(
          'reputationBadge.level.fallback',
          { level: this.resolvedLevel() },
          { defaultValue: '{{level}}' },
        )
    }
  }

  /**
   * The formatted score.
   * @returns The compact score string.
   */
  formattedScore(): string {
    return formatScore(this.score)
  }

  /**
   * The "Reputation" caption above the full variant's row.
   * @returns The caption text.
   */
  caption(): string {
    return this.i18n.t('reputationBadge.caption', {}, { defaultValue: 'Reputation' })
  }

  /**
   * The root's accessible label (score and level).
   * @returns The `aria-label`.
   */
  ariaLabel(): string {
    return this.i18n.t(
      'reputationBadge.aria',
      { score: this.formattedScore(), level: this.levelLabelText() },
      { defaultValue: '{{score}} reputation, {{level}}' },
    )
  }

  /**
   * The compact root's class list.
   * @returns The class string.
   */
  compactClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.flex({ align: 'center', gap: 'xs' }), this.className)
  }

  /**
   * The full root's class list.
   * @returns The class string.
   */
  fullClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.stack(1), this.className)
  }

  /**
   * The full variant's caption class list.
   * @returns The class string.
   */
  captionClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.textSize('xs'), cm.fontWeight('medium'))
  }

  /**
   * The full variant's row class list.
   * @returns The class string.
   */
  rowClasses(): string {
    return this.cm().flex({ align: 'center', gap: 'sm' })
  }

  /**
   * The full variant's score class list.
   * @returns The class string.
   */
  fullScoreClasses(): string {
    const cm = this.cm()
    return cm.cn(cm.textSize('lg'), cm.fontWeight('bold'))
  }

  /**
   * The compact variant's score class list.
   * @returns The class string.
   */
  compactScoreClasses(): string {
    return this.cm().cn(this.cm().fontWeight('semibold'))
  }
}
```

### Functions

#### `levelForScore(score, thresholds)`

Derive a {@link ReputationLevel} from a numeric score.

Picks the highest level whose threshold is met or exceeded. Any score
below `thresholds.contributor` returns `'newcomer'`.

```typescript
function levelForScore(score: number, thresholds?: ReputationThresholds): ReputationLevel
```

- `score` — Reputation score (any non-negative number).
- `thresholds` — Optional override thresholds; defaults to {@link DEFAULT_THRESHOLDS}.

**Returns:** The derived reputation level.

#### `colorForLevel(level)`

Return the semantic Badge color for a given level.

Unknown levels fall back to `'secondary'`.

```typescript
function colorForLevel(
  level: ReputationLevel,
): 'primary' | 'secondary' | 'success' | 'warning' | 'info'
```

- `level` — The reputation level.

**Returns:** A badge color name from `@molecule/app-ui-vue`.
