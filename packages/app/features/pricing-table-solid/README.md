## Quick Start

```tsx
import { PricingTable } from '@molecule/app-pricing-table-solid'

const checkout = (planId: string): void => {
  window.location.assign(`/checkout?plan=${planId}`)
}

export function Plans() {
  return (
    <PricingTable
      plans={[
        {
          id: 'starter',
          name: 'Starter',
          price: '$9',
          interval: '/mo',
          cta: { label: 'Get started', onClick: () => checkout('starter') },
        },
        {
          id: 'pro',
          name: 'Pro',
          price: '$29',
          interval: '/mo',
          recommended: true,
          cta: { label: 'Get started', onClick: () => checkout('pro') },
        },
      ]}
      features={[
        { label: 'Projects', values: { starter: '3', pro: 'Unlimited' } },
        { label: 'Team members', values: { starter: false, pro: true } },
      ]}
    />
  )
}
```

### Functions

#### `PricingTable(props)`

Side-by-side pricing comparison — features × plans matrix. A header row
holds plan cards (name, price, CTA); body rows show per-feature availability
(`true` → ✓, `false` → —, strings/nodes pass through). The wrapper scrolls
horizontally when columns overflow.
