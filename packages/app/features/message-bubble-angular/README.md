## Quick Start

```ts
import { Component } from '@angular/core'
import { MessageBubble, MessageComposer } from '@molecule/app-message-bubble-angular'
import type { MessageData } from '@molecule/app-message-bubble-angular'

@Component({
  selector: 'app-chat',
  standalone: true,
  imports: [MessageBubble, MessageComposer],
  template: `
    @for (message of messages; track message.id) {
      <article
        mol-message-bubble
        [message]="message"
        [isSelf]="message.author.id === 'u1'"
      ></article>
    }
    <div mol-message-composer [onSubmit]="send.bind(this)"></div>
  `,
})
export class Chat {
  messages: MessageData[] = [
    {
      id: '1',
      author: { id: 'u1', name: 'Alice', avatarSrc: '/alice.png' },
      body: 'Hey there!',
      timestamp: '10:00',
    },
    { id: '2', author: { id: 'u2', name: 'Bob' }, body: 'Hi Alice!', timestamp: '10:01' },
  ]
  send(text: string): void {
    this.messages = [...this.messages, sent(text)]
  }
}
```

```typescript
   */
  separatorContext(message: MessageData, index: number): Record<string, unknown> {
    return { $implicit: message, a0: message, a1: index, a2: undefined }
  }

  /**
   * Whether a value renders plain text.
   * @param value - The value.
   * @returns True when it is text.
   */
  hasText(value: unknown): boolean {
    return hasText(value)
  }

  /**
   * The text a value renders.
   * @param value - The value.
   * @returns The text, or `''` for a template.
   */
  textOf(value: unknown): string {
    return isTemplate(value) ? '' : textOf(value)
  }

  /**
   * Reconciles the projected empty state and the list wrapper against React's
   * branch once the view queries resolve.
   */
  ngAfterViewInit(): void {
    this.reconcile()
  }

  /**
   * Re-reconciles on later passes, so a props step that empties or refills the
   * list swaps the branch the way React does.
   */
  ngDoCheck(): void {
    this.reconcile()
  }

  /**
   * Re-reconciles after each view update: a node-valued empty state renders into
   * its outlet only during the update, so the branch decision must be read
   * afterwards.
   */
  ngAfterViewChecked(): void {
    this.reconcile()
  }

  /**
   * Applies React's condition: with no messages and an empty state, the empty
   * state renders bare and the list wrapper disappears; otherwise the wrapper
   * renders and the empty state is dropped.
   */
  private reconcile(): void {
    const empty = this.empty?.nativeElement
    const list = this.list?.nativeElement
    const hasEmpty =
      (this.messages?.length ?? 0) === 0 &&
      (isTemplate(this.emptyState) || hasText(this.emptyState) || hasNodes(empty))
    this.guard.toggle(empty, hasEmpty)
    this.guard.toggle(list, !hasEmpty)
  }
}
```

#### `ThreadIndicator`

ThreadIndicator component.

`"3 replies · last Monday"` button rendered below a message bubble. The host
IS the `<button>` (`button[mol-thread-indicator]`), matching React's root
exactly. `lastReplyAt` is text, an `ng-template`, or projected

```typescript
@Component({

  }

   * Whether a value renders plain text.
   * @param value - The value.
   * @returns True when it is text.
   */
  hasText(value: unknown): boolean {
    return hasText(value)
  }

  /**
   * The text a value renders.
   * @param value - The value.
   * @returns The text, or `''` for a template.

   * Re-reconciles on later passes, so a props change swaps the branch the way
   * React does.
   */
  ngDoCheck(): void {

  }

   * the update, so presence must be read afterwards.
   */
  ngAfterViewChecked(): void {

  private reconcile(): void {
    const span = this.lastReply?.nativeElement
    const projected = this.lastProjected?.nativeElement
    const shows = isTemplate(this.lastReplyAt) || hasText(this.lastReplyAt) || hasNodes(projected)
    this.guard.toggle(span, shows)
  }
}
```

### Runtime Dependencies

- `@angular/common`
- `@angular/core`
- `@angular/platform-browser`
- `@molecule/app-angular`
- `@molecule/app-ui`
- `@molecule/app-ui-angular`

- `isSelf` (or `MessageList.selfAuthorId`) only right-aligns the row — there is
  no built-in bubble background/tint. Style the body via `className` (or wrap
  `message.body`) if you want classic chat bubbles.
- `timestamp` renders exactly what you pass — no time-ago or date formatting.
- The composer submits on the Send button or Ctrl/Cmd+Enter (`submitOnEnter`),
  NOT plain Enter; it trims and clears on submit and blocks empty sends.
- Composer strings route through `t()` (`composer.placeholder`,
  `composer.send`) with the `@molecule/app-locales-message-bubble` companion
  bond; `ThreadIndicator` uses `thread.replies`, pluralized by `count`
  (`thread.replies_one` / `thread.replies_other`) in the same bond.
- A node-valued prop cannot be an input carrying a live DOM node: pass text,
  an `ng-template` reference, or projected content (`[slot=author]`,
  `[slot=emptyState]`, `[slot=leading]`, …).
- `MessageReactions` marks the viewer's own reaction with `aria-pressed` only —
  no visual highlight ships.
- Requires wired i18n (`useTranslation`) and a ClassMap bond; the avatar,
  button and textarea come from `@molecule/app-ui-angular`.
