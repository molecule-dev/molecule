# @molecule/app-notification-badge-angular

Angular notification badge / dot / wrapper — the counterpart of
`@molecule/app-notification-badge-react`: the same props and the same DOM
for the same input.

Exports:

- `<NotificationBadge>` (`span[mol-notification-badge]`) — count pill with

- `<NotificationDot>` (`span[mol-notification-dot]`) — tiny presence

### Classes

#### `NotificationBadge`

NotificationBadge component.

Small count pill — typically attached to a nav item, icon button, or inbox
entry. The host IS the root `<span>` (`span[mol-notification-badge]`),
matching React's root exactly. React renders null while the badge is hidden
(`hideOnZero` with a count of 0 or less); Angular's host element always
exists, so it is detached from the DOM in that state and restored when the
count shows it again.

This is the Angular counterpart of `@molecule/app-notification-badge-react`'s
`NotificationBadge`: the same props (React names), the same ClassMap tokens,
and the same DOM for the same input.

```typescript
@Component({
  selector: 'span[mol-notification-badge]',
  standalone: true,
  inputs: ['count', 'hideOnZero', 'max', 'variant', 'className'],
  host: {
    '[attr.aria-label]': 'label()',
    '[class]': 'classes()',
    '[style.min-width]': '"20px"',
    '[style.height]': '"20px"',
  },
  template: `{{ display() }}`,
})
export class NotificationBadge implements AfterViewInit, DoCheck, OnDestroy {
  /** Numeric count — renders as a pill. */
  count!: number
  /** Hide the badge when `count` is 0 (defaults to true at read time). */
  hideOnZero?: boolean
  /** When `count > max`, renders as `max+` (defaults to 99 at read time). */
  max?: number
  /** Accent color (defaults to `'error'` at read time). */
  variant?: NonNullable<NotificationBadgeProps['variant']>
  /** Extra classes. */
  className?: string

  /** The host element (the root `<span>`), for the null-render detach. */
  private readonly el = inject(ElementRef)
  /** Where the detached host lives, for the restore. */
  private detachedParent: Node | null = null
  /** The node the detached host is restored before. */
  private detachedAnchor: Node | null = null

  /**
   * The current class map.
   * @returns The UIClassMap.
   */
  cm(): UIClassMap {
    return getClassMap()
  }

  /**
   * The pill's class list (React's `cm.cn(cm.flex({…}), cm.sp('px', 1),
   * cm.roundedFull, cm.textSize('xs'), cm.fontWeight('bold'), VARIANT_BG[variant], className)`).
   * @returns The class string.
   */
  classes(): string {
    return this.cm().cn(
      this.cm().flex({ align: 'center', justify: 'center' }),
      this.cm().sp('px', 1),
      this.cm().roundedFull,
      this.cm().textSize('xs'),
      this.cm().fontWeight('bold'),
      VARIANT_BG[this.variant ?? 'error'],
      this.className,
    )
  }

  /**
   * The accessible label — React's `aria-label={`${count}`}`.
   * @returns The true count as text.
   */
  label(): string {
    return `${this.count}`
  }

  /**
   * The rendered text — React's `count > max ? `${max}+` : String(count)`.
   * @returns The display text.
   */
  display(): string {
    const max = this.max ?? 99
    return this.count > max ? `${max}+` : String(this.count)
  }

  /**
   * Whether the badge renders (React's `!(hideOnZero && count <= 0)`).
   * @returns True while the badge shows.
   */
  isVisible(): boolean {
    return !((this.hideOnZero ?? true) && this.count <= 0)
  }

  /**
   * Removes the host element from the DOM while React would render null, and
   * restores it when the badge is shown again.
   */
  syncHostVisibility(): void {
    const host = this.el.nativeElement as HTMLElement
    if (!this.isVisible()) {
      if (host.isConnected) {
        this.detachedParent = host.parentNode
        this.detachedAnchor = host.nextSibling
        host.parentNode?.removeChild(host)
      }
    } else if (this.detachedParent && !host.isConnected) {
      this.detachedParent.insertBefore(host, this.detachedAnchor)
      this.detachedParent = null
      this.detachedAnchor = null
    }
  }

  /**
   * Reconciles host visibility once the view is in the document: the first
   * ngDoCheck of a cycle runs before Angular has inserted a freshly created
   * view, so the mount-time decision is applied here.
   */
  ngAfterViewInit(): void {
    this.syncHostVisibility()
  }

  /** Re-reconciles host visibility after input writes. */
  ngDoCheck(): void {
    this.syncHostVisibility()
  }

  /** Restores the host before Angular removes it (unmount). */
  ngOnDestroy(): void {
    const host = this.el.nativeElement as HTMLElement
    // The host must be inside its parent for Angular's own removal to succeed.
    if (!host.isConnected && this.detachedParent) {
      this.detachedParent.insertBefore(host, this.detachedAnchor)
      this.detachedParent = null
      this.detachedAnchor = null
    }
  }
}
```

NotificationDot component.

#### `NotificationWrapper`

NotificationWrapper component.

Positions a `<NotificationBadge>` at a corner of any child element. The
host IS the outer `<span>` (`span[mol-notification-wrapper]`), matching
React's root exactly — the child is projected content (the default
projection) and the badge follows it inside an absolutely positioned
`<span>`. The wrapper becomes `relative` so the badge absolutely positions
correctly — wrap icon buttons, avatars, or nav entries. Same props, same DOM
as `@molecule/app-notification-badge-react`'s `NotificationWrapper`.

```typescript
@Component({
  selector: 'span[mol-notification-wrapper]',
  standalone: true,
  imports: [NotificationBadge],
  inputs: ['count', 'hideOnZero', 'variant', 'placement', 'className'],
  host: {
    '[class]': 'classes()',
    '[style.display]': '"inline-block"',
  },
  template: `<ng-content></ng-content
    ><span
      [style.position]="'absolute'"
      [style.top]="edgeValue('top')"
      [style.bottom]="edgeValue('bottom')"
      [style.right]="edgeValue('right')"
      [style.left]="edgeValue('left')"
      ><span
        mol-notification-badge
        [count]="count"
        [hideOnZero]="hideOnZeroValue()"
        [variant]="variantValue()"
      ></span
    ></span>`,
})
export class NotificationWrapper {
  /** Notification count. */
  count!: number
  /** Hide the badge when count is 0 (defaults to true at read time). */
  hideOnZero?: boolean
  /** Visual variant (defaults to `'error'` at read time). */
  variant?: NonNullable<NotificationWrapperProps['variant']>
  /** Corner placement (defaults to `'top-right'` at read time). */
  placement?: NonNullable<NotificationWrapperProps['placement']>
  /** Extra classes. */
  className?: string

  /**
   * The current class map.
   * @returns The UIClassMap.
   */
  cm(): UIClassMap {
    return getClassMap()
  }

  /**
   * The wrapper's class list (React's `cm.cn(cm.position('relative'), className)`).
   * @returns The class string.
   */
  classes(): string {
    return this.cm().cn(this.cm().position('relative'), this.className)
  }

  /**
   * The badge's `hideOnZero` (React's `hideOnZero = true` default parameter,
   * forwarded to the badge).
   * @returns The hide-on-zero flag.
   */
  hideOnZeroValue(): boolean {
    return this.hideOnZero ?? true
  }

  /**
   * The badge's `variant` (React's `variant = 'error'` default parameter,
   * forwarded to the badge).
   * @returns The variant.
   */
  variantValue(): NonNullable<NotificationWrapperProps['variant']> {
    return this.variant ?? 'error'
  }

  /**
   * One corner offset (React sets `-4` on an edge only when the placement
   * includes it).
   * @param edge - The edge to set.
   * @returns `'-4px'` when the placement includes the edge, otherwise null.
   */
  edgeValue(edge: 'top' | 'bottom' | 'right' | 'left'): string | null {
    return (this.placement ?? 'top-right').includes(edge) ? '-4px' : null
  }
}
```

Requires a wired ClassMap bond — `getClassMap()` throws before wiring.

Every variant resolves to a real theme background token, so all five are
visible in both light and dark themes: `error` / `warning` / `info` /
`success` map to the semantic status colors, and `neutral` maps to the
`surface-secondary` surface token (what `cm.surfaceSecondary` emits) for a
neutral grey fill.

Node props are projected content in Angular: the wrapper's child is its
default projection, not an input. `<NotificationWrapper>` absolutely
positions the badge 4px OUTSIDE the child's corner — an `overflow: hidden`
ancestor will clip it.
