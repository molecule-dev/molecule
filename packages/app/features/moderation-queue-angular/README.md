# @molecule/app-moderation-queue-angular

Angular moderation queue UI.

Exports `<ModerationQueue>` — a list of flagged items with kind icon,
content preview, reason chip, severity color, per-row action buttons
(approve / reject / escalate / mute), and a bulk-select toolbar (select-all
checkbox + apply-to-selected action buttons) — and the `severityColor`
helper. It is the Angular counterpart of
`@molecule/app-moderation-queue-react`: same props (React names), same DOM
for the same input.

All user-visible text is i18n'd via the companion locale bond
`@molecule/app-locales-moderation-queue`.

  onEscalate?: (id: string) => void
  onMute?: (id: string) => void
  onBulkAction?: (action: ModerationBulkAction, ids: string[]) => void
  loading?: boolean
  emptyState?: unknown
  className?: string

  private readonly internalSelected = signal<ReadonlySet<string>>(new Set())

  private readonly i18n = useTranslation()

  cm(): UIClassMap {
    return getClassMap()
  }

  selected(): ReadonlySet<string> {
    return this.internalSelected()
  }

  ids(): string[] {
    return this.items.map((item) => item.id)
  }

  allSelected(): boolean {
    const ids = this.ids()
    return ids.length > 0 && ids.every((id) => this.selected().has(id))
  }

  someSelected(): boolean {
    const ids = this.ids()
    return !this.allSelected() && ids.some((id) => this.selected().has(id))
  }

  selectionCount(): number {
    return this.ids().filter((id) => this.selected().has(id)).length
  }

  toggleOne(id: string): void {
    const next = new Set(this.selected())
    if (next.has(id)) next.delete(id)
    else next.add(id)
    this.internalSelected.set(next)
  }

  toggleAll(): void {
    const ids = this.ids()
    const every = ids.length > 0 && ids.every((id) => this.selected().has(id))
    this.internalSelected.set(every ? new Set<string>() : new Set(ids))
  }

  applyBulk(action: ModerationBulkAction): void {
    const ids = this.ids()
    const picked = Array.from(this.selected()).filter((id) => ids.includes(id))
    if (picked.length === 0) return
    this.onBulkAction?.(action, picked)
    this.internalSelected.set(new Set<string>())
  }

    this.onApprove?.(id)
