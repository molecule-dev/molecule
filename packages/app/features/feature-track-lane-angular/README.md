# @molecule/app-feature-track-lane-angular

Angular horizontal track-lane primitive for multi-track timelines (music DAW,
video editor, animation tool). Renders one lane row of draggable + resizable
clip blocks on a `pixelsPerSecond` time axis.

Angular counterpart of `@molecule/app-feature-track-lane-react`: the same
props (React names), the same DOM for the same input, and the same callback
calls, so an app switches framework by changing only the package name it
imports.

Exports `TrackLane` and the `Clip` type, plus the pure-function helpers
`clipToPixels`, `pixelsToTime` and `clampClipMove` that back the
pointer-event drag/resize math.

## API

### Interfaces

#### `Clip`

A single clip block on a track lane. `startTime` and `duration` are both in
seconds; `pixelsPerSecond` on the lane converts them to the on-screen
geometry. `id` is a stable identifier used as the row key and as the argument
passed back to all event handlers. `color` accents the body fill (defaults to
a neutral surface) and `label` is rendered inside the clip when given — text,
or a template reference for a node.

```typescript

  duration: number
  /** Optional accent color for the clip body (any valid CSS color). */
  color?: string
  /** Optional inline label rendered on top of the clip body: text or a template. */
```

#### `TrackLane`

TrackLane component.

One row of a multi-track timeline — renders draggable + resizable clip blocks
on a `pixelsPerSecond` time axis. Pointer-down on a clip body starts a
drag-to-move, pointer-down on the right-edge handle starts a drag-to-resize,
and a click without movement fires `onClipClick`. All styling routes through
`getClassMap()` and all user-visible text routes through `useTranslation()`
(via the companion `@molecule/app-locales-feature-track-lane` locale bond).

The host IS React's root `<div>` (`div[mol-track-lane]`): `role`, the
translated `aria-label`, `data-mol-id` and `data-lane` live on it.

```typescript
export class TrackLane {
  name?: string | number | TemplateRef<unknown> | null
  /** Clips on this lane, in display order. */
  clips?: Clip[]
  pixelsPerSecond?: number
  lane?: string
  height?: number
  /** Called when a clip is dragged horizontally. */
  onClipMove?: (clipId: string, startTime: number, lane?: string) => void
  /** Called when a clip's right edge handle is dragged. */
  onClipResize?: (clipId: string, duration: number, lane?: string) => void
  /** Called when a clip body is clicked (no drag). */
  onClipClick?: (clipId: string, lane?: string) => void
  selectedClipId?: string
  /** Extra classes merged onto the root element. */
  className?: string

  /** The gesture in progress, or null. */
  private drag: DragState | null = null

  readonly dragHosts = createDragHosts()

  readonly handleStyle =
    'position: absolute; top: 0px; right: 0px; bottom: 0px; width: 8px; cursor: ew-resize; touch-action: none'

    return getClassMap()
  }

  }

    return this.cm().cn(
      this.cm().flex({ align: 'center' }),
      this.cm().sp('p', 2),
      this.cm().shrink0,

  /** The lane name span's class list. */
  nameClasses(): string {
    return this.cm().cn(this.cm().textSize('sm'), this.cm().fontWeight('semibold'))

  /** A clip block's class list. */

    return this.cm().cn(this.cm().cursorPointer)
  }

  /** The clip label span's class list. */
  labelClasses(): string {
    return this.cm().cn(this.cm().textSize('xs'))
  }

  hasHeader(): boolean {
    return this.name !== undefined || this.lane !== undefined
  }

  nameTemplate(): TemplateRef<unknown> | null {
    return isTemplate(this.name) ? this.name : null
  }

  nameText(): string | null {
    return typeof this.name === 'string' || typeof this.name === 'number' ? String(this.name) : null
  }

  labelTemplate(clip: Clip): TemplateRef<unknown> | null {
    return isTemplate(clip.label) ? clip.label : null
  }

  /** The text form of one clip's label. */
  labelText(clip: Clip): string | null {
    return typeof clip.label === 'string' || typeof clip.label === 'number'

  }

    return this.selectedClipId === clip.id

    return this.i18n.t(
      'trackLane.aria.lane',
      { name: typeof this.name === 'string' ? this.name : '' },
      { defaultValue: 'Track lane {{name}}' },
    )
  }

  headerLabel(): string {
    return this.i18n.t('trackLane.header', {}, { defaultValue: 'Track' })
  }

  resizeLabel(): string {
    return this.i18n.t('trackLane.aria.resize', {}, { defaultValue: 'Resize clip' })
  }

  clipLabel(clip: Clip): string {
    return this.i18n.t(
      'trackLane.aria.clip',
      {
        label: typeof clip.label === 'string' ? clip.label : clip.id,
        startTime: clip.startTime.toFixed(2),
        duration: clip.duration.toFixed(2),
      },
      { defaultValue: 'Clip {{label}} starting at {{startTime}}s for {{duration}}s' },
    )
  }

  laneStyle(): string {
    const height = this.height ?? 44
    return [
      'position: relative',
      `height: ${height}px`,
      `min-height: ${height}px`,
      'width: 100%',
      'overflow: hidden',
      'touch-action: pan-y',
    ].join('; ')
  }

  clipStyle(clip: Clip): string {
    const { left, width } = clipToPixels(clip, this.pixelsPerSecond ?? 20)
    const selected = this.isSelected(clip)
    return [
      'position: absolute',
      `left: ${left}px`,
      `width: ${width}px`,
      'top: 4px',
      'bottom: 4px',
      `background: ${clip.color ?? 'var(--mol-track-lane-clip-bg, rgba(0, 122, 204, 0.45))'}`,

      'align-items: center',
      'padding-left: 6px',
      'padding-right: 12px',
      'overflow: hidden',
      'user-select: none',
      'touch-action: none',
    ].join('; ')
  }

  private clipDrag(clip: Clip, mode: DragMode): DragHandlers {
    const current = (): DragState | undefined =>
      this.drag?.clipId === clip.id && this.drag.mode === mode ? this.drag : undefined
    return {
      onStart: (): void => {
        if (this.drag) return
        this.drag = {
          clipId: clip.id,
          mode,
          originStartTime: clip.startTime,
          originDuration: clip.duration,
          moved: false,
        }
      },
      onMove: (dx): void => {
        const state = current()
        if (!state || (!state.moved && Math.abs(dx) < DRAG_DISTANCE_THRESHOLD_PX)) return
        state.moved = true
        const dt = pixelsToTime(Math.abs(dx), this.pixelsPerSecond ?? 20) * (dx < 0 ? -1 : 1)
        if (mode === 'move') {
          this.onClipMove?.(clip.id, clampClipMove(state.originStartTime + dt), this.lane)
        } else {
          this.onClipResize?.(
            clip.id,
            Math.max(MIN_CLIP_DURATION_SECONDS, state.originDuration + dt),
            this.lane,
          )
        }
      },
      onEnd: (dx): void => {
        const state = current()
        if (!state) return
        this.drag = null
        if (mode === 'move' && !state.moved && Math.abs(dx) < DRAG_DISTANCE_THRESHOLD_PX) {
          this.onClipClick?.(clip.id, this.lane)
        }
      },
      onCancel: (): void => {
        if (current()) this.drag = null
      },
    }
  }

  press(event: PointerEvent, clip: Clip, mode: DragMode): void {
    this.dragHosts.press(event.currentTarget as HTMLElement, event, this.clipDrag(clip, mode))
  }
}
```

### Functions

#### `clipToPixels(clip, pixelsPerSecond)`

Convert a clip to its on-lane pixel geometry.

```typescript
function clipToPixels(
  clip: { startTime: number; duration: number },
  pixelsPerSecond: number,
): { left: number; width: number }
```

- `clip` — The clip to project (`startTime` + `duration` in seconds).
- `pixelsPerSecond` — Horizontal scale; zero or negative falls back to 1.

**Returns:** `{ left, width }` in pixels.

#### `pixelsToTime(pixels, pixelsPerSecond)`

Convert a pixel offset back to a time value, clamped to `>= 0`.

```typescript
function pixelsToTime(pixels: number, pixelsPerSecond: number): number
```

- `pixels` — Offset in pixels.
- `pixelsPerSecond` — Horizontal scale; zero or negative falls back to 1.

**Returns:** Time in seconds (never negative).

#### `clampClipMove(proposedStartTime)`

Clamp the proposed new `startTime` for a clip move so the clip stays
non-negative. (No upper bound — the parent decides whether to clip to a song
length.)

```typescript
function clampClipMove(proposedStartTime: number): number
```

- `proposedStartTime` — The candidate start time in seconds.

**Returns:** The clamped start time.
