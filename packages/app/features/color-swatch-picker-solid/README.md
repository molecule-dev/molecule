## Quick Start

```tsx
import { createSignal } from 'solid-js'
import { ColorSwatchPicker } from '@molecule/app-color-swatch-picker-solid'

const swatches = [
  { value: 'red', color: '#ef4444', label: 'Red' },
  { value: 'blue', color: '#3b82f6', label: 'Blue' },
  { value: 'green', color: '#22c55e', label: 'Green' },
]

export function TagColorPicker() {
  const [selected, setSelected] = createSignal('red')
  return (
    <ColorSwatchPicker
      swatches={swatches}
      value={selected()}
      onChange={setSelected}
      ariaLabel="Tag color"
    />
  )
}
```

Grid of colored circles with single-select state. Each swatch renders as a

keys because Solid's style runtime applies each key with `setProperty`,
