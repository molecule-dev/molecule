# @molecule/app-rating-display-angular

Angular star-rating display.

Exports `RatingDisplay` — fractional star rendering (inline SVG), optional
review count tail, optional interactive mode via `onChange`. It is the
Angular counterpart of `@molecule/app-rating-display-react`: same props
(React names), same DOM for the same input.

## Quick Start

```ts
import { Component, signal } from '@angular/core'
import { RatingDisplay } from '@molecule/app-rating-display-angular'

@Component({
  selector: 'app-product-rating',
  standalone: true,
  imports: [RatingDisplay],
  template: `
    <span mol-rating-display [value]="4.5" [reviewCount]="128" (click)="scrollToReviews()"></span>
    <span mol-rating-display [value]="rating()" size="lg" [onChange]="setRating"></span>

  readonly setRating = (value: number): void => this.rating.set(value)
```

### Classes

#### `RatingDisplay`

RatingDisplay component.

Star-rating display supporting fractional values and an optional
review-count tail. When `onChange` is provided each star becomes an
interactive button. The host IS the outer `<span>`
(`span[mol-rating-display]`), matching React's root exactly, so the root's
class is a host binding; the stars sit in an inner span that carries the
rating label — `role="img"` when read-only, `role="group"` (holding the
rate buttons) in interactive mode — and the root carries no role, exactly
as React's does.

Uses inline SVG stars so the component works without icon-font fallback.
Same props (React names) and same DOM as
`@molecule/app-rating-display-react`'s `RatingDisplay`.

```typescript
@Component({
  selector: 'span[mol-rating-display]',
  standalone: true,
  inputs: [
    'value',
    'max',
    'reviewCount',
    'onReviewsClick',
    'size',
    'onChange',
    'ariaLabel',
    'className',
  ],
  host: {
    '[class]': 'classes()',
  },
  template: `<span
      [class]="starsClasses()"
      [attr.role]="onChange ? 'group' : 'img'"
      [attr.aria-label]="label()"
    >
      @for (star of stars(); track star.i) {
        @if (onChange) {
          <button
            type="button"
            (click)="select(star.i)"
            [attr.aria-pressed]="pressed(star.i) ? 'true' : 'false'"
            [attr.aria-label]="rateLabel(star.i)"
          >
            <svg [attr.width]="px()" [attr.height]="px()" viewBox="0 0 24 24" aria-hidden="true">
              <defs>
                <linearGradient [attr.id]="star.id">
                  <stop [attr.offset]="offset(star.fill)" stop-color="currentColor"></stop>
                  <stop
                    [attr.offset]="offset(star.fill)"
                    stop-color="currentColor"
                    stop-opacity="0.25"
                  ></stop>
                </linearGradient>
              </defs>
              <path [attr.d]="path" [attr.fill]="fillUrl(star.id)"></path>
            </svg>
          </button>
        } @else {
          <svg [attr.width]="px()" [attr.height]="px()" viewBox="0 0 24 24" aria-hidden="true">
            <defs>
              <linearGradient [attr.id]="star.id">
                <stop [attr.offset]="offset(star.fill)" stop-color="currentColor"></stop>
                <stop
                  [attr.offset]="offset(star.fill)"
                  stop-color="currentColor"
                  stop-opacity="0.25"
                ></stop>
              </linearGradient>
            </defs>
            <path [attr.d]="path" [attr.fill]="fillUrl(star.id)"></path>
          </svg>
        }
      }
    </span>
    @if (reviewCount !== undefined) {
      @if (onReviewsClick) {
        <button type="button" (click)="onReviewsClick($event)" [class]="reviewClasses()">
          ({{ reviewCount }})
        </button>
      } @else {
        <span [class]="textClasses()">({{ reviewCount }})</span>
      }
    }`,
})
export class RatingDisplay {
  /** Rating value (0–max). Fractional supported. */
  value!: number
  /** Maximum rating (defaults to 5 at read time). */
  max?: number
  /** Optional review count shown after the stars. */
  reviewCount?: number
  /** Called when the review-count is clicked; receives the DOM click event. */
  onReviewsClick?: (event?: MouseEvent) => void
  /** Size preset (defaults to `'md'` at read time). */
  size?: 'sm' | 'md' | 'lg'
  /** Interactive mode — star clicks call `onChange(starIndex + 1)`. */
  onChange?: (value: number) => void
  /** Accessible label override; defaults to the translated `Rating: {value} of {max}`. */
  ariaLabel?: string
  /** Extra classes. */
  className?: string
  /** The star's single SVG path. */
  readonly path = STAR_PATH

  /** The i18n handle (locale-reactive, so the labels follow a switch). */
  private readonly i18n = useTranslation()

  /**
   * The current class map.
   * @returns The UIClassMap.
   */
  cm(): UIClassMap {
    return getClassMap()
  }

  /**
   * The effective maximum (React's default parameter: a later-bound write of
   * `undefined` must not erase the `5` default).
   * @returns The maximum rating.
   */
  maxValue(): number {
    return this.max ?? 5
  }

  /**
   * The star size in pixels (React's `SIZE_PX[size]`, which is `undefined` for
   * an unknown size so both axes are omitted).
   * @returns The pixel size.
   */
  px(): number | undefined {
    return SIZE_PX[this.size ?? 'md']
  }

  /**
   * One render state per star: its index, its fractional fill and its gradient
   * id (minted per render, as React mints one per star per render).
   * @returns The star states.
   */
  stars(): StarState[] {
    return Array.from({ length: this.maxValue() }, (_value, i) => ({
      i,
      fill: Math.max(0, Math.min(1, this.value - i)),
      id: starId(),
    }))
  }

  /**
   * The gradient stop offset for a fill fraction.
   * @param fill - Star fill fraction (0–1).
   * @returns The percentage offset.
   */
  offset(fill: number): string {
    return `${fill * 100}%`
  }

  /**
   * The `<path>` fill referencing the star's gradient.
   * @param id - The star's gradient id.
   * @returns The `url(#id)` fill.
   */
  fillUrl(id: string): string {
    return `url(#${id})`
  }

  /**
   * The per-star button label in interactive mode.
   * @param i - Star index (0-based).
   * @returns The accessible label.
   */
  rateLabel(i: number): string {
    return this.i18n.t(
      'ratingDisplay.aria.rate',
      { value: i + 1, max: this.maxValue() },
      { defaultValue: 'Rate {{value}} of {{max}}' },
    )
  }

  /**
   * Calls `onChange` with the rating a clicked star selects.
   * @param i - Star index (0-based).
   */
  select(i: number): void {
    this.onChange?.(i + 1)
  }

  /**
   * Whether the star at index `i` is pressed: it is within the rounded rating.
   * @param i - Star index (0-based).
   * @returns The button's `aria-pressed` state.
   */
  pressed(i: number): boolean {
    return i + 1 <= Math.round(this.value)
  }

  /**
   * The root `<span>` class list (`cm.flex({ align: 'center', gap: 'xs' })` +
   * `className`).
   * @returns The class string.
   */
  classes(): string {
    return this.cm().cn(this.cm().flex({ align: 'center', gap: 'xs' }), this.className)
  }

  /**
   * The inner stars span class (`cm.flex({ align: 'center', gap: 'xs' })`, the
   * same list as the root so wrapping the stars does not move the layout).
   * @returns The class string.
   */
  starsClasses(): string {
    return this.cm().flex({ align: 'center', gap: 'xs' })
  }

  /**
   * The root `aria-label` (`ariaLabel`, or the translated default).
   * @returns The accessible label.
   */
  label(): string {
    return (
      this.ariaLabel ??
      this.i18n.t(
        'ratingDisplay.aria.rating',
        { value: this.value, max: this.maxValue() },
        { defaultValue: 'Rating: {{value}} of {{max}}' },
      )
    )
  }

  /**
   * The review-count button class (`cm.textSize('sm')` + `cm.link`).
   * @returns The class string.
   */
  reviewClasses(): string {
    return this.cm().cn(this.cm().textSize('sm'), this.cm().link)
  }

  /**
   * The review-count text span class (`cm.textSize('sm')`).
   * @returns The class string.
   */
  textClasses(): string {
    return this.cm().textSize('sm')
  }
}
```

The host element IS the outer `<span>` (`span[mol-rating-display]`), exactly
as in React. Stars are inline SVG filled with `currentColor` — set a text
color on the wrapper (e.g. a warning/amber token) to tint them; unfilled
portions render at 25% opacity of the same color. Accessible labels are
translated through `useTranslation()`, matching the React package and
following a locale switch: the container label (`ratingDisplay.aria.rating`,
overridable via `ariaLabel`) and the per-star "Rate N of M" button labels in
interactive mode (`ratingDisplay.aria.rate`). Requires a wired ClassMap bond.
