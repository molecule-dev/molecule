# @molecule/app-settings-actions-bar-solid

Solid sticky Save/Cancel bar with "Saved" timestamp + loading state for
settings and form pages.

Exports `SettingsActionsBar`. Props: `onSave` (required; sync or async),
`onCancel?` (Cancel hidden when omitted), `loading?` (disables Save and
shows "Saving…"), `disabled?`, `savedAt?` (epoch ms → "Saved 3m ago"),
`error?` (inline node), `sticky?` (default true), `leading?` (node before
the status), `className?`.

## Quick Start

```tsx
import { createSignal } from 'solid-js'

import { SettingsActionsBar } from '@molecule/app-settings-actions-bar-solid'

export function ProfileActions(props: { save: () => Promise<void>; reset: () => void }) {
  const [saving, setSaving] = createSignal(false)
  const [savedAt, setSavedAt] = createSignal<number | null>(null)
  return (
    <SettingsActionsBar
      onSave={async () => {
        setSaving(true)
        try {
          await props.save()
          setSavedAt(Date.now())
        } finally {
          setSaving(false)
        }
      }}
      onCancel={props.reset}
      loading={saving()}
      savedAt={savedAt()}
    />
  )
}
```
