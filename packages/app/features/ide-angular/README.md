```typescript
  text: string
  /** An optional action button (or buttons): a link (`href`) and/or a click handler. */

   * in sequence — plain strings as text, {@link ChatEventCardAction}s as inline underlined
   * links — so prose and links interleave freely (e.g. text → an inline link → a trailing
   * period). When set, the renderer uses this INSTEAD of `text` + appended `action`s, so a
   * link can sit mid-sentence rather than only at the end. Segments carry their own spacing
   * (no auto-space is inserted between them). Keep `text` populated with a plain-text
   * equivalent for accessibility / non-toned consumers. Only honored when `tone` is set.
   */

   * When true, ChatPanel renders the card as a stand-out tip box rather than muted inline
   * text. Prefer setting {@link ChatEventCard.tone} (which implies emphasis AND picks the
   * accent colour + icon); `emphasized` without a `tone` falls back to the neutral `info`
   * tone. The app opts in; the shared package never infers emphasis from a card's copy.
   */
  emphasized?: boolean
  /**
   * The card's tip TONE — picks its accent colour + default icon so every notice card
   * shares ONE consistent box (icon + tinted body + a uniform 1px border + actions),
   * differing only by colour/icon per kind:

   * - `gold` — amber, lightbulb (an honest tip / onboarding note)
   * - `upgrade` — amber, clock (a plan/limit/budget nudge)
   * - `success` — green, check (a completed action, e.g. a saved script)
   * - `signup` — primary, sign-in (an auth nudge)
   *
   * Setting `tone` implies emphasis. Cards that supply composable {@link ChatEventCard.content}
   * render their inline links in the box; cards that supply `action`(s) render them as a
   * consistent row of accent buttons. Omit `tone` (and `emphasized`) for a plain muted line.
   */
  tone?: 'info' | 'gold' | 'upgrade' | 'success' | 'signup'
  /**
   * Optional icon-name override (a `@molecule/app-icons` glyph) — defaults to the tone's
   * icon. Use only a name that exists in the bonded set (`getIcon` throws otherwise);
   * sets with extra glyphs register them via `CustomIconNames` augmentation.
   */
  icon?: IconName
  /**
   * The limit this card already explains, named by the same `limitType` the backend puts
   * on its limit errors (e.g. `'ai_cost'`). A limit is hit ONCE but can surface twice —

   * limit banner when the NEXT send is refused — which reads as two cards saying the same
   * thing. When a live error carries the same `limitType`, ChatPanel hides this card for
   * as long as that banner is up, so exactly one surface states the limit; the card
   * returns as soon as the error clears. The banner is never the one suppressed: it is
   * the only feedback the refused send gets, and its message can be more specific than
   * the card's (a platform-capacity refusal shares `limitType` with a personal-budget
   * one). The app owns the identifier; the shared package only matches it.
   */
  coversLimitType?: string
  /**

   * the work (a refused deploy, a blocked outbound connection). A critical card
   * keeps its timestamp even while the viewer has timestamps turned off. The app
   * decides; the shared package never infers criticality from a card's tone or copy.

  critical?: boolean
```

#### `AutoCommitAction`

Actions the countdown reducer accepts.

- `set` — apply a `/autocommit <seconds>` command (`seconds <= 0` disables);
  arms AND starts a fresh countdown (an explicit, just-now user choice).
- `hydrate` — restore a cadence persisted on the project (e.g. on reload),
  enabled but PAUSED (`seconds <= 0` disables). Unlike `set`, it does NOT
  start counting down — the countdown only re-arms on the next file change —
  so reopening a project never auto-commits a tree the user hasn't touched.
- `reset` — a file changed; restart the full countdown (no-op when disabled).
- `tick` — one second elapsed; decrement toward zero (no-op when paused).
- `fired` — a commit was just dispatched; pause until the next file change.

```typescript
type AutoCommitAction =
  | { type: 'set'; seconds: number }
  | { type: 'hydrate'; seconds: number }
  | { type: 'reset' }
  | { type: 'tick' }
  | { type: 'fired' }
```

```typescript
  /** Name of the glyph to look up in the bonded icon set (e.g. `'sync'`). */
  name!: IconName

  /** Width and height of the rendered SVG in pixels. Defaults to 16. */
  size = 16

  /** Class name forwarded to the root `<svg>`. */
  className?: string

  /**
   * A click handler forwarded to the root `<svg>` — React's rest spread
   * accepts any SVG/HTML attribute, and a handler reaches this host the same
   * way a host listener.
   */
  onClick?: (event: MouseEvent) => void

  /** The resolved icon data of the current `name`. */
  private resolved?: IconData

  /** The raw-markup child nodes this component created (so only they are removed). */
  private rawNodes: ChildNode[] = []

  /** The svg host element. */
  private readonly host: ElementRef<SVGSVGElement> = inject(ElementRef)

  /**
   * Resolves the new glyph and rebuilds the raw-markup children. Angular's
   * template branch destroys and creates the `<path>` views on its own; only
   * the raw branch's children are managed here.
   */
  ngOnChanges(): void {
    const icon = getIcon(this.name)
    this.resolved = icon
    const el = this.host.nativeElement
    // The raw children are this component's own nodes (parsed from the icon
    // set, never rendered by Angular): drop exactly those and nothing else, so
    // the framework's `<path>` views and comment anchors stay untouched.
    for (const node of this.rawNodes) {
      // eslint-disable-next-line molecule-local/no-hand-placed-nodes -- raw markup this component imported (rawNodes), not a node Angular rendered
      node.remove()
    }
    this.rawNodes = []
    if (icon.svg) {
      const doc = new DOMParser().parseFromString(
        `<svg xmlns="${SVG_NAMESPACE}">${icon.svg}</svg>`,
        'image/svg+xml',
      )
      for (const child of Array.from(doc.documentElement.childNodes)) {
        const imported = document.importNode(child, true) as ChildNode
        el.appendChild(imported)
        this.rawNodes.push(imported)
      }
    }
  }

  /**
   * The rendered size (the default applies when a later-bound input writes
   * `undefined` over the field default, matching React's absent-prop default).
   * @returns The pixel size.
   */
  protected attrSize(): number {
    return this.size ?? 16
  }

  /**
   * The `viewBox` attribute value, with React's fallback.
   * @returns The viewBox.
   */
  protected viewBox(): string {
    return this.resolved?.viewBox || '0 0 20 20'
  }

  /**
   * The `fill` attribute value (`none` for raw-markup glyphs without an
   * explicit fill and for stroked glyphs, matching React's two branches).
   * @returns The fill.
   */
  protected fillAttr(): string {
    const icon = this.resolved
    if (!icon) return 'currentColor'
    if (icon.svg) return icon.fill || 'none'
    return icon.stroke ? 'none' : icon.fill || 'currentColor'
  }

  /**

   * React's absent attributes are absent here too).

    strokeWidth: number | null
    strokeLinecap: string | null
    strokeLinejoin: string | null

      return { stroke: null, strokeWidth: null, strokeLinecap: null, strokeLinejoin: null }

      strokeWidth: icon.strokeWidth ?? null,
      strokeLinecap: icon.strokeLinecap ?? null,
      strokeLinejoin: icon.strokeLinejoin ?? null,
    }
  }

   * The glyph paths (empty for raw-markup glyphs — React's raw branch has no
   * `<path>` children).
```

```typescript
   * The rows QuickPicker renders: the `/workspace/` prefix stripped for
   * display, the bare name as the label, the display path as the detail beside
   * it when it differs, each with the row icon template (React's `items`
   * memo — one identity until the file list or the icon template changes, so a
   * change-detection pass alone never re-sets the picker's highlight).

      return {
        id: file,
        label: name,
        detail: display !== name ? display : undefined,
        icon,
      }
    })

  protected readonly i18n = useTranslation()

   * re-fetch the list (`ngOnChanges` fires for the first binding too).
   * @param changes - The changed inputs.

    if (changes['projectId']) this.load()

   * Resolves the row icon template once the view exists; the rows recompute
   * through their `computed` when it arrives.
```

```typescript
    el.setPointerCapture(event.pointerId)
    document.body.style.cursor = this.horizontal ? 'col-resize' : 'row-resize'
    document.body.style.userSelect = 'none'
  }

  /** Drags: every move calls `onResize` with the pixels moved since the last move (a move that lands where the last one did is no move at all). */
  protected onPointerMove(event: PointerEvent): void {
    if (!this.draggingLive) return
    const current = this.horizontal ? event.clientX : event.clientY
    const delta = current - this.start
    if (delta === 0) return
    this.start = current
    this.onResize(delta)
  }

  /** Ends a drag (release or cancel), lets the capture go and restores the page the way the drag start found it. */
  protected onPointerUp(event: PointerEvent): void {
    this.endDrag(event)
  }

  /** Ends a drag (release or cancel), lets the capture go and restores the page the way the drag start found it. */
  protected onPointerCancel(event: PointerEvent): void {
    this.endDrag(event)
  }

  /** The shared drag end: drops the guard and the highlight, lets the capture go and restores the page. */
  private endDrag(event: PointerEvent): void {
    if (!this.draggingLive) return
    this.draggingLive = false
    this.dragging = false
    const el = event.currentTarget as HTMLDivElement
    if (el.hasPointerCapture(event.pointerId)) el.releasePointerCapture(event.pointerId)
    document.body.style.cursor = ''
    document.body.style.userSelect = ''
  }

  /** Keyboard resize: the arrows nudge along the handle's axis (left/up shrink, right/down grow); every other key passes through untouched. */
  protected onKeyDown(event: KeyboardEvent): void {
    const decreaseKey = this.horizontal ? 'ArrowLeft' : 'ArrowUp'
    const increaseKey = this.horizontal ? 'ArrowRight' : 'ArrowDown'
    if (event.key === decreaseKey) {
      event.preventDefault()
      this.onResize(-KEYBOARD_STEP_PX)
    } else if (event.key === increaseKey) {
      event.preventDefault()
      this.onResize(KEYBOARD_STEP_PX)
    }
  }

  /** Highlights the line while the pointer is over the handle (a crossing into the wrapped line is not a leave — see the module note). */
  protected onPointerOver(event: PointerEvent): void {
    const el = event.currentTarget as HTMLDivElement
    if (event.relatedTarget instanceof Node && el.contains(event.relatedTarget)) return
    this.hovered = true
  }

  /** Restores the line once the pointer has actually left the handle. */
  protected onPointerOut(event: PointerEvent): void {
    const el = event.currentTarget as HTMLDivElement
    if (event.relatedTarget instanceof Node && el.contains(event.relatedTarget)) return
    this.hovered = false
  }
}
```

```typescript
  selector: 'div[mol-workspace-layout]',
  standalone: true,
  imports: [ResizeHandle],
  hostDirectives: [MolPlacement],
  inputs: ['className'],
  host: {
    'data-workspace-layout': '',
    '[class]': 'rootClass()',
    '[attr.style]': 'rootStyle()',
  },
  // prettier-ignore
  template: `@for (index of indexes(); track index) {
    <div #panel [class]="panelClass()" [attr.style]="panelStyle(index)"></div>
    @if (index < indexes().length - 1) {
      <div mol-resize-handle [direction]="'horizontal'" [value]="handleValue(index)" [onResize]="resizeFor(index)"></div>
    }
  }<span #panes><ng-content /></span>`,
})
export class WorkspaceLayout {
  /** Extra classes from the host, appended to the root's ClassMap classes. */
  className?: string

  /** The bonded ClassMap (styling tokens resolve at render time). */
  protected readonly cm: UIClassMap = getClassMap()

  /** The IDE workspace state and its layout actions. */
  private readonly workspace = useWorkspace()

  /** The current panel layout, read reactively by the template. */
  protected readonly layout = this.workspace.layout

  /** How many panes render — known once the projected content exists. */
  private readonly childCount = signal(0)

  /** One entry per pane; the template renders a wrapper (and divider) per entry. */
  protected readonly indexes = computed(() =>
    Array.from({ length: this.childCount() }, (_, index) => index),
  )

  /** The visible panel configs taken one per rendered pane, in order. */
  private readonly panelConfigs = computed(() =>
    this.layout()
      .panels.filter((panel) => panel.visible !== false)
      .slice(0, this.childCount()),
  )

  /** The wrapper element the template rendered for each pane, in order. */
  @ViewChildren('panel') private panelEls!: QueryList<ElementRef<HTMLElement>>

  /** The capture of the projected panes; it is out of the page, and each pane renders in its wrapper. */
  @ViewChild('panes', { static: true }) private readonly panesCapture?: ElementRef<HTMLElement>

  /** Per-divider resize callbacks, kept stable across change-detection passes. */
  private readonly resizers = new Map<number, (delta: number) => void>()

  /**
   * The root's classes: the row, full width and height, and the host's extra
   * classes, composed exactly as React composes them.
   *
   * @returns The resolved class string.
   */
  protected rootClass(): string {
    return this.cm.cn(
      this.cm.flex({ direction: 'row' }),
      this.cm.w('full'),
      this.cm.h('full'),
      this.className,
    )
  }

  /**
   * The root's inline style: clipping at full height, as React writes it.
   *
   * @returns The `style` attribute text.
   */
  protected rootStyle(): string | null {
    return styleText({ overflow: 'hidden', minHeight: 0 })
  }

  /**
   * A pane wrapper's classes: the centered column React gives it.
   *
   * @returns The resolved class string.
   */
  protected panelClass(): string {
    return this.cm.flex({ direction: 'col' })
  }

  /**
   * A pane's inline geometry: its size as the flex basis, and whether it is
   * the trailing pane that absorbs the leftover space.
   *
   * @param index - The pane's index in the row.
   * @returns The `style` attribute text.
   */
  protected panelStyle(index: number): string {
    const size = livePanelSize(this.layout(), this.panelConfigs(), index)
    const grow = index === this.indexes().length - 1 ? 1 : 0
    return `flex: ${grow} ${grow} ${size}%; min-width: 0px; overflow: hidden; position: relative`
  }

  /**
   * The leading pane's share (percent) of the two panes divider `index`
   * separates — the value the focusable separator reports as `aria-valuenow`.
   *
   * @param index - The divider's index (the pane before it).
   * @returns The leading pane's share of the pair, 0–100.
   */
  protected handleValue(index: number): number {
    const configs = this.panelConfigs()
    const lead = livePanelSize(this.layout(), configs, index)
    const trail = livePanelSize(this.layout(), configs, index + 1)
    const total = lead + trail
    return total > 0 ? (lead / total) * 100 : 50
  }

  /**
   * The divider's resize callback for one pane index (stable between passes).
   *
   * @param index - The divider's index (the panel it follows).
   * @returns A function taking the pixels moved since the last move.
   */
  protected resizeFor(index: number): (delta: number) => void {
    let handler = this.resizers.get(index)
    if (!handler) {
      handler = (delta: number): void => this.handleResize(index, delta)
      this.resizers.set(index, handler)
    }
    return handler
  }

  /**
   * Resizes the panel before divider `index` by a pixel delta: reads the
   * panel's live size, clamps the change against the layout container's
   * measured width (1000 when the DOM reports none) and writes the new
   * percentage back through the workspace provider.
   *
   * @param index - The divider's index (the panel it follows).
   * @param delta - The pixels moved since the last move.
   */
  private handleResize(index: number, delta: number): void {
    const configs: PanelConfig[] = this.panelConfigs()
    const config = configs[index]
    if (!config) return
    const containerWidth = document.querySelector('[data-workspace-layout]')?.clientWidth || 1000
    const currentSize = livePanelSize(this.layout(), configs, index)
    const newSize = clampPanelSize(currentSize, delta, containerWidth)
    this.workspace.resizePanel(config.id, newSize)
  }

  /**
   * Reads the slot's elements (React's `Children.toArray(children).filter(isValidElement)`),
   * renders one wrapper per element, and renders each element inside its wrapper. An element
   * already placed is listed by the placeholder that holds its place in the slot, so it is found
   * again through it; a text child is taken out of the page, as React drops it.
   *
   * @param hasContent - Reads the slot's children.
   */
  private placePanes(hasContent: SlotReader): void {
    const capture = this.panesCapture?.nativeElement
    if (!capture) return
    setElementPresent(capture, false)
    const panes: Node[] = []
    for (const node of hasContent.nodes(capture)) {
      const child = heldElement(node) ?? node
      if (child.nodeType === 1) panes.push(child)
      // React drops a text child (it is not an element).
      else setElementPresent(child, false)
    }
    if (this.childCount() !== panes.length) this.childCount.set(panes.length)
    const wrappers = this.panelEls?.map((ref) => ref.nativeElement) ?? []
    // The wrappers for a new count render in the next pass, which places the panes again.
    if (wrappers.length !== panes.length) return
    panes.forEach((pane, index) => placeElement(pane, wrappers[index] ?? null))
  }

  constructor() {
    usePlacement((hasContent) => this.placePanes(hasContent))
  }
}
```

#### `TestRunProjection`

Incremental card state. A server id is applied once, in order; completed runs are compacted
while per-file verdicts survive. No raw event history is retained or replayed into a fold.

```typescript
class TestRunProjection {
  private readonly runs = new Map<string, ProjectedRun>()
  private readonly results: Record<string, TestResultEntry> = {}
  private readonly pending = new Map<string, string>()
  private lastEventId: string | null = null
  private order = 0

  /** Apply one event and return the visible state. Missing ids support older custom hosts only. */
  apply(event: TestRunFeedEvent, activeOwnRunId: string | null = null): TestsRunState {
    if (event.eventId) {
      if (this.lastEventId && compareEventIds(event.eventId, this.lastEventId) <= 0)
        return this.show(activeOwnRunId)
      this.lastEventId = event.eventId
    }
    const order = ++this.order
    const previous = this.runs.get(event.runId) ?? {
      state: { ...EMPTY_RUN_STATE, runId: event.runId, running: event.type !== 'done' },
      covered: new Set<string>(),
      startOrder: order,
      lastOrder: order,
    }
    if (!previous.state.running && event.type === 'done') return this.show(activeOwnRunId)

    const id =
      event.type === 'case' || event.type === 'result' || event.type === 'output'
        ? event.id
        : undefined
    const covered =
      event.type === 'start' ? event.ids : id != null && !previous.covered.has(id) ? [id] : []
    for (const file of covered) {
      previous.covered.add(file)
      delete this.results[file]
      this.pending.set(file, event.runId)
    }

    let next = applyTestRunEvent(previous.state, event)
    if (event.type === 'start' && event.startedAt) {
      const startedAt = Date.parse(event.startedAt)
      if (Number.isFinite(startedAt)) next = { ...next, startedAt }
    }
    if (event.type === 'result') {
      const verdict = next.results[event.id]
      if (verdict) this.results[event.id] = verdict
      // The latest verdict is what counts, even when an older run produced it after a newer start.
      this.pending.delete(event.id)
    }
    if (event.type === 'done') {
      for (const [file, runId] of this.pending) {
        if (runId === event.runId) this.pending.delete(file)
      }
    }
    this.runs.set(event.runId, { ...previous, state: next, lastOrder: order })
    this.compact()
    return this.show(activeOwnRunId)
  }

  /** Keep every live run (including the active own run) and only the latest completed run. */
  private compact(): void {
    const completed = [...this.runs.values()]
      .filter((run) => !run.state.running)
      .sort((a, b) => b.lastOrder - a.lastOrder)
    for (const run of completed.slice(1)) {
      if (run.state.runId) this.runs.delete(run.state.runId)
    }
  }

  /** Render current per-file state, with progress and controls selected independently. */
  show(activeOwnRunId: string | null = null): TestsRunState {
    const activeOwn = activeOwnRunId ? this.runs.get(activeOwnRunId) : undefined
    const running = [...this.runs.values()]
      .filter((run) => run.state.running)
      .sort((a, b) => a.startOrder - b.startOrder || a.lastOrder - b.lastOrder)
    const latest = [...this.runs.values()].sort((a, b) => a.lastOrder - b.lastOrder).at(-1)
    const shown = activeOwn?.state.running
      ? activeOwn
      : (running.at(-1) ?? latest ?? { state: EMPTY_RUN_STATE })
    const runningIds = new Set(
      [...this.pending]
        .filter(([, runId]) => this.runs.get(runId)?.state.running === true)
        .map(([id]) => id),
    )
    const own = shown.state.runId != null && shown.state.runId === activeOwnRunId
    const currentId =
      shown.state.currentId && runningIds.has(shown.state.currentId) ? shown.state.currentId : null
    const cases = Object.fromEntries(
      Object.entries(shown.state.cases).filter(([id]) => runningIds.has(id)),
    )
    return {
      ...shown.state,
      currentId,
      cases,
      runningIds: [...runningIds],
      results: { ...this.results },
      external: shown.state.runId == null ? undefined : !own,
      skipUnavailable: own ? shown.state.skipUnavailable : true,
    }
  }
}
```

the union: adding a setting to {@link SETTINGS} without adding its case here
is a compile error, so a setting can never ship with no translation site.
Each `defaultValue` equals that setting's `label` field, so the English
output is identical either way.

```typescript
function settingLabel(t: TranslateFunction, id: SettingKey): string
```

- `t` — The caller's translate function (React passes `useTranslation()`'s `t`).
- `id` — The setting id.

**Returns:** The localized label.

#### `settingDescription(t, id, values)`

The localized one-line explanation of a user-controllable setting (the
`ide.chat.settings.<id>.desc` keys), through the CALLER's translate function.
One literal `t()` call per {@link SettingKey}, exhaustive over the union (see
{@link settingLabel}). Each `defaultValue` equals that setting's `description`
field, so the English output is identical either way; the descriptions
interpolate `{{agentName}}`, filled from the caller's agent identity.

```typescript
function settingDescription(
  t: TranslateFunction,
  id: SettingKey,
  values?: InterpolationValues,
): string
```

- `t` — The caller's translate function.
- `id` — The setting id.
- `values` — Interpolation values (the descriptions interpolate `{{agentName}}`).

**Returns:** The localized description.

#### `USER_ACCENT_STYLE`

User-message accent stripe. The left edge of a real user message draws a vertical,
