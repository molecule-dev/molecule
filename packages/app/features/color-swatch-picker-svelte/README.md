```svelte

  let selected = $state('red')

  const swatches = [
    { value: 'red', color: '#ef4444', label: 'Red' },
    { value: 'blue', color: '#3b82f6', label: 'Blue' },
    { value: 'green', color: '#22c55e', label: 'Green' },
```
