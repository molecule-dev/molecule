# @molecule/app-pricing-page-angular

Angular drop-in pricing page used by every paid molecule.dev flagship app.

Components:
`<PricingPage />` — Renders one card per tier, with optional
monthly/yearly toggle and Stripe Checkout
CTA. Reads tiers from `usePricingTiers()`
unless an explicit `tiers` prop is passed.

`<PlanUpdatedPage />` — Post-checkout success page rendered at
Stripe's `success_url` redirect target.

It is the Angular counterpart of `@molecule/app-pricing-page-react`: the
same exports, the same inputs (React prop names) and the same DOM for the
same input.

Translations live in the companion locale bond
`@molecule/app-locales-pricing-page`. The checkout transport lives in
`@molecule/app-billing-angular` (`usePricingTiers()` calls
`GET /api/billing/tiers`; `useStartCheckout()` posts to
`/api/billing/checkout`) — your API must serve those routes and the HTTP
client must be wired. NOTE: `@molecule/app-billing-angular` also exports
its own different pricing page (compound LimitsList/LimitsItem layout);
import from the package whose API you are using. The default checkout
redirects with `window.location.assign(checkoutUrl)` — pass `onCheckout`
to route through an SPA router or a non-Stripe flow. `<PlanUpdatedPage />`
here is the pricing-page-flavored success page; a standalone plan-updated
page lives in `@molecule/app-plan-updated-page-angular` — import the one
whose kit you are using.

   * projected content landed, React would not render the CTA at all, so the

    const el = this.ctaLabelEl?.nativeElement
    if (el) this.ctaProjected.set(hasRenderableContent(el))

   * Whether a text CTA label input renders (a projected label is not).
   *
   * @returns True when the label renders as text.

  get hasCtaLabel(): boolean {
    return this.ctaLabel !== undefined && this.ctaLabel !== null && this.ctaLabel !== ''
  }

  /**
   * Whether the CTA renders: a truthy string input or a projected
   * `[slot=ctaLabel]` element.
   *
   * @returns True when the CTA shows.
   */
  get showCta(): boolean {
    return this.hasCtaLabel || this.ctaProjected()
  }

  /**
   * Passes the click event to the caller's handler, as React's
   * `onClick={onCta}` does.
   *
   * @param event - The DOM click event.

  fireCta(event: unknown): void {

export class PricingPage implements OnInit, PricingPageProps {
  /** Optional pre-fetched tiers. */
  tiers?: PricingTierEntry<unknown>[]

  /** Optional override for the checkout handler. */
  onCheckout?: (tier: PricingTierEntry<unknown>, price: PricingTierPrice) => void | Promise<void>

  /** When true (default), renders a monthly/yearly toggle. */
  showPeriodToggle?: boolean

  /** Default billing period for the toggle. */
  defaultPeriod?: PricingPagePeriod

  renderLimits?: TemplateRef<unknown> | null

  /** Optional translation key for the page heading. */
  headingKey?: string

  /** Optional English fallback for the heading. */
  headingDefault?: string

  className?: string

  private readonly i18n = useTranslation()

  /** The translation function for the template. */
  protected t = this.i18n.t

  /** The bonded ClassMap (styling tokens resolve at render time). */
  protected readonly cm: UIClassMap = getClassMap()

  protected readonly tiersState = usePricingTiers<unknown>()

  /** The checkout action and its async state. */
  protected readonly startState = useStartCheckout()

  protected period: PricingPagePeriod = 'month'

  ngOnInit(): void {
    this.period = this.defaultPeriod ?? 'month'
  }

  usingFetched(): boolean {
    return this.tiers === undefined
  }

  tiersList(): PricingTierEntry<unknown>[] | null {
    return this.tiers ?? this.tiersState.data()?.data ?? null
  }

  /** Whether the loading line shows (React's first early return). */
  showLoading(): boolean {
    return this.usingFetched() && this.tiersState.loading() && !this.tiersList()
  }

  /** Whether the error line shows (React's second early return). */
  showError(): boolean {
    return this.usingFetched() && (this.tiersState.error() != null || !this.tiersList())
  }

  molId(): string {
    if (this.showLoading()) return 'pricing-page-loading'
    if (this.showError()) return 'pricing-page-error'
    return 'pricing-page'
  }

  /** Whether any tier carries a yearly price (the toggle needs one). */
  hasYearly(): boolean {
    return (this.tiersList() ?? []).some((tier) => tier.prices.some((p) => p.period === 'year'))
  }

  priceFor(tier: PricingTierEntry<unknown>): PricingTierPrice | undefined {
    return tier.prices.find((p) => p.period === this.period) ?? tier.prices[0]
  }

  isObjectLimits(limits: unknown): limits is Record<string, unknown> {
    return limits != null && typeof limits === 'object'
  }

  /** The entries of a limits object, in insertion order. */
  entries(limits: Record<string, unknown>): [string, unknown][] {
    return Object.entries(limits)
  }

  limitValue(value: unknown): string {
    if (typeof value === 'boolean') return value ? '✓' : '—'
    if (typeof value === 'number') {
      if (value >= 999_999) return '∞'
      return value.toLocaleString()
    }
    if (value == null) return '—'
    if (typeof value === 'string') return value
    try {
      return JSON.stringify(value)
    } catch (_error) {
      return String(value)
    }
  }

  limitsContext(limits: unknown): { $implicit: unknown; a0: unknown } {
    return { $implicit: limits, a0: limits }
  }

  fireCheckout(tier: PricingTierEntry<unknown>): void {
    const price = this.priceFor(tier)
    if (price) void this.handleCheckout(tier, price)
  }

  private async handleCheckout(
    tier: PricingTierEntry<unknown>,
    price: PricingTierPrice,
  ): Promise<void> {
    if (!price.stripePriceId) return
    if (this.onCheckout) {
      await this.onCheckout(tier, price)
      return
    }
    const response = await this.startState.start(price.stripePriceId)
    if (response?.checkoutUrl && typeof window !== 'undefined') {
      window.location.assign(response.checkoutUrl)
    } else if (response?.updated && typeof window !== 'undefined') {
      window.location.reload()
    }
  }
}
