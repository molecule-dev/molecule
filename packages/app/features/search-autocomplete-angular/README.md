# @molecule/app-search-autocomplete-angular

Search input with a typeahead suggestion popover.

Exports `<SearchAutocomplete>` (an attribute-selector component:
`div[mol-search-autocomplete]`, whose host is React's root `<div>`) and the
`SuggestionItem<T>` type (`{ id, label, meta?, data? }`). Two data modes —
pass ONE of: `onSearch(query)` (async fetch, debounced; default 200 ms) or
`suggestions` (caller-controlled list). Other props: `value` / `onChange`
(controlled input), `onSelect(item)`, `placeholder?` (defaults to the
translated `search.placeholder`), `debounceMs?`, `minChars?` (default 1),
`className?`.

Angular counterpart of `@molecule/app-search-autocomplete-react`: the same
exports, the same props (React names) and the same DOM for the same input.

## Quick Start

```ts
import { Component, signal } from '@angular/core'
import { SearchAutocomplete, type SuggestionItem } from '@molecule/app-search-autocomplete-angular'

@Component({
  selector: 'app-user-search',
  standalone: true,
  imports: [SearchAutocomplete],
  template: `
    <div
      mol-search-autocomplete
      [value]="query()"
      [onChange]="setQuery"
      [onSearch]="fetchUsers"
      [onSelect]="pick"
      placeholder="Search users…"
    ></div>
  `,
})
export class UserSearch {
  readonly query = signal('')
  readonly setQuery = (next: string): void => this.query.set(next)
  readonly fetchUsers = (q: string): Promise<SuggestionItem<{ id: string }>[]> =>
    fetch(`/api/users?q=${encodeURIComponent(q)}`).then((r) => r.json())
  readonly pick = (item: SuggestionItem<{ id: string }>): void => console.log(item.data?.id)
}
```

```typescript
export class SearchAutocomplete implements OnInit, OnChanges, OnDestroy {
  /** Current input value. */
  value = ''
  onChange?: (value: string) => void
  onSearch?: (query: string) => Promise<SuggestionItem[]> | SuggestionItem[]
  suggestions?: SuggestionItem[]
  onSelect?: (item: SuggestionItem) => void
  placeholder?: string
  ariaLabel?: string
  debounceMs?: number
  minChars?: number
  className?: string

  @ViewChild('inputEl') private inputEl?: ElementRef<HTMLInputElement>

  private readonly i18n = useTranslation()

  private readonly cdr = inject(ChangeDetectorRef)
  private readonly host = inject(ElementRef<HTMLElement>)

  private readonly internal = signal<SuggestionItem[]>([])
  private readonly open = signal(false)
  private readonly activeIndex = signal(-1)
  private request = 0
  private timer: ReturnType<typeof setTimeout> | undefined
  private readonly baseId = `mol-search-autocomplete-${nextGeneratedId++}`

  readonly listId = `${this.baseId}-listbox`

  readonly listStyle = LIST_STYLE

  ngOnInit(): void {
    document.addEventListener('mousedown', this.onDoc)
  }

  ngOnChanges(changes: SimpleChanges): void {
    if ('value' in changes) this.activeIndex.set(-1)
    if (
      'value' in changes ||
      'onSearch' in changes ||
      'debounceMs' in changes ||
      'minChars' in changes
    )
      this.scheduleSearch()
  }

  ngOnDestroy(): void {
    document.removeEventListener('mousedown', this.onDoc)
    clearTimeout(this.timer)
  }

  /** The current class map. */
  cm(): UIClassMap {
    return getClassMap()
  }

  field(): ReturnType<typeof fieldLayout> {
    return fieldLayout(this.cm())
  }

  classes(): string {
    return this.cm().cn(this.cm().position('relative'), this.className)
  }

  inputClasses(): string {
    return this.cm().input({ error: false, size: 'md', align: undefined })
  }

  /** The listbox's class list. */
  listClasses(): string {
    return this.cm().cn(this.cm().stack(0), this.cm().sp('p', 1))
  }

  labelClasses(): string {
    return this.cm().fontWeight('medium')
  }

  metaClasses(): string {
    return this.cm().textSize('xs')
  }

  optionClasses(index: number): string {
    return this.cm().cn(
      this.cm().sp('px', 2),
      this.cm().sp('py', 2),
      this.cm().textSize('sm'),
      this.cm().cursorPointer,
      index === this.active() && this.cm().surfaceSecondary,
    )
  }

  get inputId(): string {
    return `${this.baseId}-input`
  }

  optionId(index: number): string {
    return `${this.baseId}-option-${index}`
  }

  placeholderText(): string {
    return this.placeholder ?? this.i18n.t('search.placeholder', {}, { defaultValue: 'Search…' })
  }

  ariaLabelText(): string {
    return this.ariaLabel ?? this.i18n.t('search.aria.label', undefined, { defaultValue: 'Search' })
  }

  list(): SuggestionItem[] {
    return this.suggestions ?? this.internal()
  }

  isExpanded(): boolean {
    return this.open() && this.list().length > 0
  }

  active(): number {
    const index = this.activeIndex()
    return index < this.list().length ? index : -1
  }

  activeDescendant(): string | null {
    return this.isExpanded() && this.active() >= 0 ? this.optionId(this.active()) : null
  }

  isTemplate(value: unknown): value is TemplateRef<unknown> {
    return isTemplate(value)
  }

  asTemplate(value: unknown): TemplateRef<unknown> {
    return value as TemplateRef<unknown>
  }

  textOf(value: unknown): string {
    return textOf(value)
  }

  private debounce(): number {
    return this.debounceMs ?? 200
  }

  private threshold(): number {
    return this.minChars ?? 1
  }

  private scheduleSearch(): void {
    clearTimeout(this.timer)
    const search = this.onSearch
    const query = this.value
    if (!search || query.length < this.threshold()) {
      this.internal.set([])
      return
    }
    this.timer = setTimeout(() => {
      const token = ++this.request
      Promise.resolve(search(query))
        .then((result) => {
          if (token === this.request) this.internal.set(result)
          this.cdr.markForCheck()
        })
        .catch((_error) => {
          if (token === this.request) this.internal.set([])
          this.cdr.markForCheck()
        })
    }, this.debounce())
  }

  handleInput(event: Event): void {
    this.onChange?.((event.target as HTMLInputElement).value)
    this.open.set(true)
    const input = this.inputEl?.nativeElement
    if (input && input.value !== String(this.value)) input.value = String(this.value)
  }

  handleFocus(): void {
    if (this.value.length >= this.threshold()) this.open.set(true)
  }

  handleKeydown(event: KeyboardEvent): void {
    const current = this.list()
    if (event.key === 'ArrowDown') {
      event.preventDefault()
      if (!this.open()) {
        if (current.length > 0) this.open.set(true)
        return
      }
      if (current.length === 0) return
      this.activeIndex.update((index) => (index + 1) % current.length)
    } else if (event.key === 'ArrowUp') {
      event.preventDefault()
      if (!this.open()) {
        if (current.length > 0) this.open.set(true)
        return
      }
      if (current.length === 0) return
      this.activeIndex.update((index) => (index <= 0 ? current.length - 1 : index - 1))
    } else if (event.key === 'Enter') {
      const index = this.active()
      if (this.isExpanded() && index >= 0) {
        event.preventDefault()
        this.select(current[index] as SuggestionItem)
      }
    } else if (event.key === 'Escape') {
      if (this.open()) {
        event.preventDefault()
        this.open.set(false)
        this.activeIndex.set(-1)
      }
    }
  }

  onOptionEnter(index: number): void {
    this.activeIndex.set(index)
  }

  onOptionDown(event: MouseEvent, suggestion: SuggestionItem): void {
    event.preventDefault()
    this.select(suggestion)
  }

  private select(item: SuggestionItem): void {
    this.onSelect?.(item)
    this.open.set(false)
    this.activeIndex.set(-1)
  }

  private readonly onDoc = (event: MouseEvent): void => {
    if (!this.open()) return
    if (this.host.nativeElement.contains(event.target as Node)) return
    this.open.set(false)
    this.activeIndex.set(-1)
    this.cdr.markForCheck()
  }
}
```

- Throws unless provided the binding's molecule providers (`@molecule/app-angular`)
  with a bonded ClassMap and i18n provider.
- Full keyboard support: it's an ARIA combobox — ArrowDown/ArrowUp move the
  highlight (opening the popover), Enter selects the highlighted option, and
  Escape closes. `aria-activedescendant` tracks the active `role="option"`.
- Stale-response safe: out-of-order `onSearch` responses are dropped (each
  request carries a token; only the latest applies), so a slow earlier fetch
  can't overwrite fresher results. `onSearch` needs no cancellation of its own.
- The popover surface uses `var(--color-surface)` with a WHITE fallback
  and black-alpha borders — define `--color-surface` (dark themes) or the
  dropdown stays light.
- Default placeholder translations: `@molecule/app-locales-search-autocomplete`.
